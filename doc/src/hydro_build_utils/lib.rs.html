7#[macro_export]
8macro_rules! emit_nightly_configuration {
9    () => {
10        println!("cargo:rerun-if-env-changed=RUSTC_BOOTSTRAP");
11        println!("cargo::rustc-check-cfg=cfg(nightly)");
12        if matches!(
13            $crate::rustc_version::version_meta().map(|meta| meta.channel),
14            Ok($crate::rustc_version::Channel::Nightly)
15        ) || option_env!("RUSTC_BOOTSTRAP") == Some("1")
16        {
17            println!("cargo:rustc-cfg=nightly");
18        }
19    };
20}

22#[cfg(feature = "insta")]
23#[macro_export]
24macro_rules! nightly_wrapper {
25    ($statement:stmt) => {
26        $crate::insta::with_settings!({
27            prepend_module_to_snapshot => option_env!("CARGO_TARGET_TMPDIR").is_some(), // Only for integration tests.
28            snapshot_path => if cfg!(nightly) { "snapshots-nightly" } else { "snapshots" },
29            filters => vec![
30                // Stageleft quote macro names embed the source line/column of the `q!` macro
31                // invocation (e.g. `__stageleft_quote_src_cluster_paxos_rs_157_21`), which churns
32                // snapshots whenever unrelated code moves. Redact the position for stability.
33                (r"(__stageleft_quote_\w+_rs)_\d+_\d+", "${1}_LINE_COL"),
34            ],
35        }, {
36            $statement;
37        });
38    }
39}

57#[cfg(feature = "insta")]
58/// Like [`nightly_wrapper!`], but without any [`filters`](insta::Settings::set_filters) applied.
59///
60/// Useful for snapshots that are sensitive to the exact code expanded into the test body (e.g.
61/// backtrace tests, where the extra `filters` setup code perturbs frame symbolization).
62#[macro_export]
63macro_rules! nightly_wrapper_unfiltered {
64    ($statement:stmt) => {
65        $crate::insta::with_settings!({
66            prepend_module_to_snapshot => option_env!("CARGO_TARGET_TMPDIR").is_some(), // Only for integration tests.
67            snapshot_path => if cfg!(nightly) { "snapshots-nightly" } else { "snapshots" },
68        }, {
69            $statement;
70        });
71    }
72}

83#[cfg(feature = "trybuild")]
84#[macro_export]
85macro_rules! trybuild_compile_fail {
86    ($glob:expr) => {{
87        let source_dir = std::path::Path::new("tests/compile-fail");
88        let stderr_dir = source_dir.join(if cfg!(nightly) { "nightly" } else { "stable" });
89
90        // Remove any existing .rs files/symlinks in the channel dir first.
91        // Stale broken symlinks (e.g. from rebasing) break trybuild.
92        for entry in std::fs::read_dir(&stderr_dir).unwrap().flatten() {
93            if entry.path().extension().and_then(|e| e.to_str()) == Some("rs") {
94                let _ = std::fs::remove_file(entry.path());
95            }
96        }
97        // Symlink all .rs files from the source directory into the channel-specific
98        // stderr directory so trybuild can find them next to the .stderr files.
99        for entry in std::fs::read_dir(source_dir).unwrap().flatten() {
100            let path = entry.path();
101            if path.extension().and_then(|e| e.to_str()) == Some("rs") {
102                let dest = stderr_dir.join(entry.file_name());
103                let original = std::path::Path::new("..").join(entry.file_name());
104                #[cfg(unix)]
105                std::os::unix::fs::symlink(&original, &dest).unwrap();
106                #[cfg(windows)]
107                std::os::windows::fs::symlink_file(&original, &dest).unwrap();
108            }
109        }
110
111        let pattern = stderr_dir.join($glob);
112        let t = $crate::trybuild::TestCases::new();
113        t.compile_fail(pattern.to_str().unwrap());
114    }};
115}
