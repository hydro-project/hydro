6/// Given an _unbounded_ input stream, emits values arbitrarily split into batches over multiple iterations in the same order.
7///
8/// `batch_eager()` is one of three loop-ingress ("windowing") operators, which differ only in
9/// whether they cause the surrounding `loop { ... }` to fire:
10/// - `batch()` triggers the loop only when its windowed input is non-empty.
11/// - `batch_lazy()` never triggers the loop on its own; its data is only observed if the loop
12///   fires for some other reason (otherwise dropped at tick end).
13/// - `batch_eager()` **always** triggers the loop, even when the windowed input is empty.
14///
15/// Because it forces the loop body to run, `batch_eager()` is only valid at the entry of a
16/// root-level loop. It is disallowed inside nested loops, where forcing the loop to always fire
17/// would prevent the fixpoint iteration from terminating.
18///
19/// Note that `batch_eager()` only forces the loop body to run when a tick executes; it does not
20/// schedule additional ticks on its own (unlike `spin()`).
21pub const BATCH_EAGER: OperatorConstraints = OperatorConstraints {
22    name: "batch_eager",
23    categories: &[OperatorCategory::Windowing],
24    hard_range_inn: RANGE_1,
25    soft_range_inn: RANGE_1,
26    hard_range_out: RANGE_1,
27    soft_range_out: RANGE_1,
28    num_args: 0,
29    persistence_args: RANGE_0,
30    type_args: RANGE_0,
31    is_external_input: false,
32    flo_type: Some(FloType::WindowingEager),
33    ports_inn: None,
34    ports_out: None,
35    input_delaytype_fn: |_| None,
36    // Same as batch() — identity in inline codegen. The loop-gate logic (see `emit_loop_gate`)
37    // uses the `WindowingEager` flo type to force the loop to fire unconditionally.
38    write_fn: |wc @ &WriteContextArgs { .. }, _diagnostics| {
39        let write_iterator = identity_write_iterator_fn(wc);
40        Ok(OperatorWriteOutput {
41            write_iterator,
42            ..Default::default()
43        })
44    },
45};
