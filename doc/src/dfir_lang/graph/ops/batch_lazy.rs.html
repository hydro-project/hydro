6/// Given an _unbounded_ input stream, emits values arbitrarily split into batches over multiple iterations in the same order.
7///
8/// Unlike `batch()`, this operator does NOT cause the loop to fire. If the loop
9/// does not fire for another reason (e.g., a non-lazy `batch` or `defer_tick`
10/// has data), the lazy-batched data is available. If the loop never fires that
11/// tick, the data is simply dropped (reclaimed by the bump allocator at tick end).
12///
13/// `batch_lazy()` is one of the loop-ingress ("windowing") operators, which differ in whether
14/// they cause the surrounding `loop { ... }` to fire and whether pending data is retained:
15/// - `batch()` triggers the loop only when its windowed input is non-empty.
16/// - `batch_lazy()` never triggers the loop on its own; its data is only observed if the loop
17///   fires for some other reason (otherwise dropped at tick end).
18/// - `batch_eager()` always triggers the loop, even when the windowed input is empty.
19/// - `snapshot()` never triggers the loop on its own, but retains the latest value
20///   (last-write-wins) across ticks and emits it every time the loop fires.
21/// - `snapshot_keyed()` never triggers the loop on its own, but retains the latest value *per
22///   key* (from `(K, Option<V>)` updates) across ticks and emits every entry every time the loop
23///   fires.
24/// - `snapshot_prefix()` never triggers the loop on its own, but retains every element received
25///   so far across ticks and replays the whole prefix every time the loop fires.
26pub const BATCH_LAZY: OperatorConstraints = OperatorConstraints {
27    name: "batch_lazy",
28    categories: &[OperatorCategory::Windowing],
29    hard_range_inn: RANGE_1,
30    soft_range_inn: RANGE_1,
31    hard_range_out: RANGE_1,
32    soft_range_out: RANGE_1,
33    num_args: 0,
34    persistence_args: RANGE_0,
35    type_args: RANGE_0,
36    is_external_input: false,
37    flo_type: Some(FloType::WindowingLazy),
38    ports_inn: None,
39    ports_out: None,
40    input_delaytype_fn: |_| None,
41    // Same as batch() — identity in inline codegen.
42    write_fn: |wc @ &WriteContextArgs { .. }, _diagnostics| {
43        let write_iterator = identity_write_iterator_fn(wc);
44        Ok(OperatorWriteOutput {
45            write_prologue: Default::default(),
46            write_iterator,
47            write_iterator_after: Default::default(),
48            write_tick_end: Default::default(),
49        })
50    },
51};
