8/// > 1 input stream, 1 output stream
9///
10/// > Arguments: two arguments, both closures. The first closure is used to create the initial
11/// > value for the accumulator, and the second is used to combine new items with the existing
12/// > accumulator value. The second closure takes two two arguments: an `&mut Accum` accumulated
13/// > value, and an `Item`.
14///
15/// Akin to Rust's built-in [`fold`](https://doc.rust-lang.org/std/iter/trait.Iterator.html#method.fold)
16/// operator, except that it takes the accumulator by `&mut` instead of by value. Folds every item
17/// into an accumulator by applying a closure, returning the final result.
18///
19/// > Note: The closures have access to the [`context` object](surface_flows.mdx#the-context-object).
20///
21/// `fold` can also be provided with one generic lifetime persistence argument, either
22/// `'tick` or `'static`, to specify how data persists. With `'tick`, Items will only be collected
23/// within the same tick. With `'static`, the accumulated value will be remembered across ticks and
24/// will be aggregated with items arriving in later ticks. When not explicitly specified
25/// persistence defaults to `'tick`.
26///
27/// ```dfir
28/// // should print `Reassembled vector [1,2,3,4,5]`
29/// source_iter([1,2,3,4,5])
30///     -> fold::<'tick>(Vec::new, |accum: &mut Vec<_>, elem| {
31///         accum.push(elem);
32///     })
33///     -> assert_eq([vec![1, 2, 3, 4, 5]]);
34/// ```
35pub const FOLD: OperatorConstraints = OperatorConstraints {
36    name: "fold",
37    categories: &[OperatorCategory::Fold],
38    hard_range_inn: RANGE_1,
39    soft_range_inn: RANGE_1,
40    hard_range_out: &(0..=1),
41    soft_range_out: &(0..=1),
42    num_args: 2,
43    persistence_args: &(0..=1),
44    type_args: RANGE_0,
45    is_external_input: false,
46    flo_type: None,
47    ports_inn: None,
48    ports_out: None,
49    input_delaytype_fn: |_| None,
50    write_fn: |wc @ &WriteContextArgs {
51                   root,
52                   op_span,
53                   work_fn_async,
54                   ident,
55                   is_pull,
56                   inputs,
57                   outputs,
58                   arguments,
59                   ..
60               },
61               diagnostics| {
62        let init_fn = &arguments[0];
63        let func = &arguments[1];
64        let singleton_output_ident = wc.make_ident("singleton_output");
65
66        let initializer_func_ident = wc.make_ident("initializer_func");
67        let init = quote_spanned! {op_span=>
68            (#initializer_func_ident)()
69        };
70
71        let [persistence] = wc.persistence_args(diagnostics);
72
73        let input = &inputs[0];
74        let accumulator_ident = wc.make_ident("accumulator");
75        let item_ident = wc.make_ident("item");
76
77        // The initializer closure is evaluated once, in the prologue.
78        let mut write_prologue = quote_spanned! {op_span=>
79            #[allow(unused_mut, reason = "for if `Fn` instead of `FnMut`.")]
80            let mut #initializer_func_ident = #init_fn;
81        };
82        // For `'static` the accumulator state lives in the prologue, persisting across ticks.
83        // For `'tick` the accumulator is created fresh each tick (within `write_iterator`), so it
84        // can be moved (not cloned) into the output.
85        if Persistence::Static == persistence {
86            write_prologue.extend(quote_spanned! {op_span=>
87                #[allow(clippy::redundant_closure_call)]
88                let mut #singleton_output_ident = #init;
89            });
90        }
91        let make_tick_state = matches!(persistence, Persistence::Tick).then(|| {
92            quote_spanned! {op_span=>
93                #[allow(clippy::redundant_closure_call)]
94                let mut #singleton_output_ident = #init;
95            }
96        });
97
98        let assign_accum_ident = quote_spanned! {op_span=>
99            #[allow(unused_mut)]
100            let mut #accumulator_ident = &mut #singleton_output_ident;
101        };
102        let foreach_body = quote_spanned! {op_span=>
103            #[inline(always)]
104            fn call_comb_type<Accum, Item>(
105                accum: &mut Accum,
106                item: Item,
107                mut func: impl FnMut(&mut Accum, Item),
108            ) {
109                (func)(accum, item);
110            }
111            #[allow(clippy::redundant_closure_call)]
112            call_comb_type(&mut *#accumulator_ident, #item_ident, #func);
113        };
114
115        let write_iterator = if is_pull {
116            match persistence {
117                // `'tick`: owned-mode fold. The accumulator is created fresh this tick and the
118                // output pull emits it by move (not cloned).
119                Persistence::Tick => quote_spanned! {op_span=>
120                    #[allow(clippy::redundant_closure_call)]
121                    let #singleton_output_ident = #init;
122                    // Eagerly consumes the input to ensure updated state.
123                    let #ident = #work_fn_async(
124                        #root::dfir_pipes::pull::Pull::fold(
125                            #input,
126                            #singleton_output_ident,
127                            #func,
128                        )
129                    ).await;
130                },
131                // `'static`: borrowed-mode fold. The accumulator persists in the prologue state,
132                // and the output pull emits a clone of its current value.
133                Persistence::Static => quote_spanned! {op_span=>
134                    let #ident = {
135                        // Eagerly consumes the input to ensure updated state.
136                        let __accumulated = #work_fn_async(
137                            #root::dfir_pipes::pull::Pull::fold_ref(
138                                #input,
139                                &mut #singleton_output_ident,
140                                #func,
141                            )
142                        ).await;
143                        #root::dfir_pipes::pull::Pull::map(
144                            __accumulated,
145                            |__val: &mut _| ::std::clone::Clone::clone(&*__val),
146                        )
147                    };
148                },
149            }
150        } else if outputs.is_empty() {
151            // Terminal push: fold is a singleton reference target with no downstream.
152            quote_spanned! {op_span=>
153                #make_tick_state
154                let #ident = #root::dfir_pipes::push::for_each(|#item_ident| {
155                    #assign_accum_ident
156
157                    #foreach_body
158                });
159            }
160        } else if Persistence::Tick == persistence {
161            // `'tick`: owned-mode fold. The accumulator is created fresh each tick and moved
162            // (not cloned) into the output on finalize.
163            let output = &outputs[0];
164            quote_spanned! {op_span=>
165                let #ident = {
166                    #[inline(always)]
167                    fn __push_fold_owned<Acc, Item, CombFn, Next>(
168                        acc: Acc,
169                        comb_fn: CombFn,
170                        next: Next,
171                    ) -> #root::dfir_pipes::push::Accumulate<
172                        #root::dfir_pipes::push::FoldState<Acc, CombFn, Acc, Item>,
173                        Next,
174                    >
175                    where
176                        CombFn: ::std::ops::FnMut(&mut Acc, Item),
177                        Next: #root::dfir_pipes::push::Push<Acc, ()>,
178                    {
179                        #root::dfir_pipes::push::fold(acc, comb_fn, next)
180                    }
181                    #[allow(clippy::redundant_closure_call)]
182                    let #singleton_output_ident = #init;
183                    __push_fold_owned(
184                        #singleton_output_ident,
185                        #func,
186                        #output,
187                    )
188                };
189            }
190        } else {
191            // `'static`: borrowed-mode fold. The accumulator persists in the prologue state, so
192            // emit a clone of its current value on finalize.
193            let output = &outputs[0];
194            quote_spanned! {op_span=>
195                let #ident = {
196                    #[inline(always)]
197                    fn __push_fold<'a, Acc, Item, CombFn, Next>(
198                        acc_ref: &'a mut Acc,
199                        comb_fn: CombFn,
200                        next: Next,
201                    ) -> #root::dfir_pipes::push::Accumulate<
202                        #root::dfir_pipes::push::FoldState<&'a mut Acc, CombFn, Acc, Item>,
203                        Next,
204                    >
205                    where
206                        CombFn: ::std::ops::FnMut(&mut Acc, Item),
207                        Next: #root::dfir_pipes::push::Push<&'a mut Acc, ()>,
208                    {
209                        #root::dfir_pipes::push::fold(acc_ref, comb_fn, next)
210                    }
211                    __push_fold(
212                        &mut #singleton_output_ident,
213                        #func,
214                        #root::dfir_pipes::push::map(
215                            |__val: &mut _| ::std::clone::Clone::clone(&*__val),
216                            #output,
217                        ),
218                    )
219                };
220            }
221        };
222
223        Ok(OperatorWriteOutput {
224            write_prologue,
225            write_iterator,
226            ..Default::default()
227        })
228    },
229};
