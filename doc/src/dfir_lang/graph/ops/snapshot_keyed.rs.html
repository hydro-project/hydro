6/// > Input: `(K, Option<V>)` updates. Output: `(K, V)` entries.
7///
8/// The keyed counterpart of `snapshot()`: retains the **latest** value *per key* across ticks
9/// and emits every current entry (as clones) each time the loop fires, without causing the loop
10/// to fire.
11///
12/// Each input item is an update to one key: `(k, Some(v))` sets `k`'s current value to `v`
13/// (replacing any previous value), and `(k, None)` removes `k`. The resulting map is a persistent
14/// view held across ticks: it is **not consumed** when the loop fires, so if the loop fires again
15/// without any new updates, the same entries are observed again. Keys and values must therefore
16/// implement `Clone` (and keys `Eq + Hash`). Before any update has arrived, `snapshot_keyed()`
17/// emits nothing.
18///
19/// Use `snapshot_keyed()` to observe keyed state (e.g. a keyed aggregation) from inside a loop.
20/// A plain `snapshot()` holds only a single item, so it would retain just the most recent entry
21/// of a keyed feed. For feeds without removals, map each entry to an upsert first:
22/// `-> map(|(k, v)| (k, Some(v))) -> snapshot_keyed()`.
23///
24/// Like `snapshot()`, `snapshot_keyed()` is only valid at the entry of a root-level loop: nested
25/// loops iterate to fixpoint within a single tick, so cross-tick retention does not apply to them.
26///
27/// Memory note: the retained map holds at most one value per (non-removed) key.
28///
29/// ```dfir
30/// trigger = source_iter([()]);
31/// updates = source_iter([("a", Some(1)), ("b", Some(2)), ("a", Some(3)), ("b", None), ("c", Some(4))]);
32/// loop {
33///     trigger -> batch() -> for_each(std::mem::drop);
34///     // Observes the latest value per key, with `"b"` removed: `("a", 3)` and `("c", 4)`.
35///     updates -> snapshot_keyed() -> for_each(|(k, v)| println!("{}: {}", k, v));
36/// };
37/// ```
38pub const SNAPSHOT_KEYED: OperatorConstraints = OperatorConstraints {
39    name: "snapshot_keyed",
40    categories: &[OperatorCategory::Windowing],
41    hard_range_inn: RANGE_1,
42    soft_range_inn: RANGE_1,
43    hard_range_out: RANGE_1,
44    soft_range_out: RANGE_1,
45    num_args: 0,
46    persistence_args: RANGE_0,
47    type_args: RANGE_0,
48    is_external_input: false,
49    flo_type: Some(FloType::WindowingRetain),
50    ports_inn: None,
51    ports_out: None,
52    input_delaytype_fn: |_| None,
53    // Identity in inline codegen, like `snapshot()`. The retention behavior comes from the
54    // handoff feeding this operator being declared outside the tick closure as a persistent
55    // keyed map (rather than a tick-local bump-allocated vec) that applies `(K, Option<V>)`
56    // updates, and that the loop reads (cloning every entry, not consuming the map) each time it
57    // fires. See `DfirGraph::as_code_with_options`.
58    write_fn: |wc @ &WriteContextArgs { .. }, _diagnostics| {
59        let write_iterator = identity_write_iterator_fn(wc);
60        Ok(OperatorWriteOutput {
61            write_prologue: Default::default(),
62            write_iterator,
63            write_iterator_after: Default::default(),
64            write_tick_end: Default::default(),
65        })
66    },
67};
