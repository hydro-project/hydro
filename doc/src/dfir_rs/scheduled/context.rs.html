22/// Coordinates waking between [`Context`] (inside the tick closure) and [`Dfir`]
23/// (the external runner). Shared via `Arc` between both.
24///
25/// When external data arrives (e.g., a tokio stream receives a message), the [`Context::waker`]
26/// fires, which sets `can_start_tick` and wakes the [`Dfir::run`](Dfir::run) task so it starts a new tick.
27/// Implements [`Wake`] directly so it can be used as a `Waker` without an extra wrapper.
28#[doc(hidden)]
29pub struct WakeState {
30    /// Set to `true` when external data arrives, signaling that a new tick should run.
31    /// Checked by [`Dfir::run_tick`](Dfir::run_tick) and [`Dfir::run_available`](Dfir::run_available).
32    can_start_tick: std::sync::atomic::AtomicBool,
33    /// Wakes the [`Dfir::run`](Dfir::run) task from its idle `poll_fn` sleep.
34    task_waker: futures::task::AtomicWaker,
35}
36
37impl Default for WakeState {
38    fn default() -> Self {
39        Self {
40            can_start_tick: std::sync::atomic::AtomicBool::new(false),
41            task_waker: futures::task::AtomicWaker::new(),
42        }
43    }
44}
45
46impl Wake for WakeState {
47    fn wake(self: Arc<Self>) {
48        self.wake_by_ref();
49    }
50
51    fn wake_by_ref(self: &Arc<Self>) {
52        self.can_start_tick.store(true, Ordering::Relaxed);
53        self.task_waker.wake();
54    }
55}

57/// A lightweight context for inline codegen that avoids the overhead of the full
58/// scheduled graph (no tokio channels, no scheduler queues, no loop machinery).
59///
60/// Exposes methods that operator-generated code calls on both
61/// `df` (for prologues: `request_task`) and
62/// `context` (for iterators: `current_tick`, `schedule_subgraph`, etc.).
63///
64/// # Metrics typestate
65///
66/// The `M` type parameter is a [`MetricsState`] typestate which statically records whether this
67/// context tracks runtime metrics:
68///
69/// - `Context<DfirMetrics>` (the default) is created via [`Context::new_with_metrics`] and holds
70///   a shared `Rc<DfirMetrics>` handle, accessible via [`Context::metrics`].
71/// - `Context<()>` is created via [`Context::new`] and holds no metrics state whatsoever. The
72///   `metrics()` accessor does not exist on it, so it is a compile error to try to read metrics
73///   from a context which never tracked them.
74pub struct Context<M: MetricsState = DfirMetrics> {
75    /// Counter for number of ticks run.
76    current_tick: TickInstant,
77    /// Coordinates waking between [`Context`] (inside the tick closure) and [`Dfir`]
78    /// (the external runner). Shared via `Arc` between both. Implements [`Wake`].
79    wake_state: Arc<WakeState>,
80    /// Live-updating DFIR runtime metrics via interior mutability (`Rc<DfirMetrics>`), or `()`
81    /// when metrics are not tracked.
82    metrics: M::Handle,
83    /// Tasks buffered via [`Self::request_task`], spawned by [`Dfir::spawn_tasks`]
84    /// once the runtime is running inside a tokio `LocalSet`.
85    #[cfg(feature = "tokio")]
86    tasks_to_spawn: Vec<Pin<Box<dyn Future<Output = ()> + 'static>>>,
87}
88
89impl Context<()> {
90    /// Create a new inline context with a fresh wake state and **no** metrics tracking.
91    ///
92    /// The returned `Context<()>` statically contains no metrics state. Use
93    /// [`Context::new_with_metrics`] to create a `Context<DfirMetrics>` which tracks metrics.
94    pub fn new() -> Self {
95        Self::with_metrics_handle(())
96    }
97}
98
99impl Default for Context<()> {
100    fn default() -> Self {
101        Self::new()
102    }
103}

105impl Context<DfirMetrics> {
106    /// Create a new inline context with a fresh wake state and the given (pre-populated) metrics.
107    pub fn new_with_metrics(metrics: DfirMetrics) -> Self {
108        Self::with_metrics_handle(Rc::new(metrics))
109    }
110
111    /// Returns a reference to the runtime metrics.
112    pub fn metrics(&self) -> &Rc<DfirMetrics> {
113        &self.metrics
114    }
115}

117impl<M: MetricsState> Context<M> {
118    fn with_metrics_handle(metrics: M::Handle) -> Self {
119        Self {
120            current_tick: TickInstant::default(),
121            wake_state: Arc::default(),
122            metrics,
123            #[cfg(feature = "tokio")]
124            tasks_to_spawn: Vec::new(),
125        }
126    }
127
128    // --- Methods called as `df.xxx()` in operator prologues ---
129
130    /// Buffers an async task to be spawned later by `Dfir::spawn_tasks`.
131    ///
132    /// Tasks are deferred because `write_prologue` runs during graph construction,
133    /// which may occur before a tokio `LocalSet` is entered. Buffered tasks are
134    /// drained and spawned via `tokio::task::spawn_local` at the start of
135    /// [`Dfir::run_tick`]. Tasks requested after that point remain buffered until
136    /// the next call to [`Dfir::run_tick`].
137    #[cfg(feature = "tokio")]
138    pub fn request_task<Fut>(&mut self, future: Fut)
139    where
140        Fut: Future<Output = ()> + 'static,
141    {
142        self.tasks_to_spawn.push(Box::pin(future));
143    }
144
145    // --- Methods called as `context.xxx()` in operator iterators ---
146
147    /// Gets the current tick count.
148    pub fn current_tick(&self) -> TickInstant {
149        self.current_tick
150    }
151
152    /// Signals that external data has arrived and a new tick should be started.
153    pub fn schedule_subgraph(&self, is_external: bool) {
154        if is_external {
155            self.wake_state.wake_by_ref();
156        }
157    }
158
159    /// Returns a waker that signals external data has arrived.
160    pub fn waker(&self) -> std::task::Waker {
161        std::task::Waker::from(self.wake_state.clone())
162    }
163
164    /// Increments the tick counter.
165    /// Called by the generated tick closure at the end of each tick.
166    #[doc(hidden)]
167    pub fn __end_tick(&mut self) {
168        self.current_tick += crate::scheduled::ticks::TickDuration::SINGLE_TICK;
169    }
170}

172/// An executable DFIR dataflow, as created by
173/// [`dfir_syntax!`](crate::dfir_syntax). Provides the [`Self::run`],
174/// [`Self::run_available`], and [`Self::run_tick`] family of methods to
175/// execute the dataflow.
176///
177/// # Design
178///
179/// The inline codegen generates an `async move |df: &mut Context<M>|` closure that captures
180/// dataflow-specific state (handoff buffers, source iterators) and receives the [`Context`]
181/// (tick counter, metrics) by reference each tick. `Dfir` owns both the
182/// closure and the context, and coordinates tick lifecycle and idle/wake behavior.
183///
184/// We use a single opaque closure rather than generating a bespoke struct per dataflow because:
185/// - The closure naturally captures exactly the state it needs with correct lifetimes
186/// - No codegen needed for struct definitions, field accessors, or initialization
187/// - Rust's async closure machinery handles the complex state machine (suspend/resume across
188///   `.await` points) that would be very difficult to replicate in a generated struct
189///
190/// The `Tick` type parameter is bounded by [`TickClosure`] (not `AsyncFnMut` directly) to
191/// support type erasure via [`TickClosureErased`] / [`DfirErased`] for heterogeneous
192/// collections (e.g., the sim runtime storing multiple locations in a `Vec`). The concrete
193/// (non-erased) path used by trybuild and embedded has zero overhead.
194///
195/// # Metrics
196///
197/// Whether this `Dfir` tracks runtime metrics is determined statically by
198/// [`TickClosure::Metrics`] (see [`MetricsState`]). The [`Self::metrics`] and
199/// [`Self::metrics_intervals`] accessors only exist when `Tick::Metrics = DfirMetrics`. When
200/// returning a `Dfir` from a function via `impl TickClosure`, use
201/// `Dfir<impl TickClosure<Metrics = DfirMetrics>>` if callers need to access the metrics.
202pub struct Dfir<Tick: TickClosure> {
203    /// Async closure which runs a single tick when called.
204    tick_closure: Tick,
205    /// Coordinates waking between [`Context`] (inside the tick closure) and [`Dfir`]
206    /// (the external runner). Shared via `Arc` between both. Implements [`Wake`].
207    wake_state: Arc<WakeState>,
208    /// The inline context, owned by `Dfir` and passed to the tick closure by reference.
209    context: Context<Tick::Metrics>,
210    /// See [`Self::meta_graph()`].
211    #[cfg(feature = "meta")]
212    meta_graph: Option<DfirGraph>,
213    /// See [`Self::diagnostics()`].
214    #[cfg(feature = "meta")]
215    diagnostics: Option<Vec<Diagnostic<SerdeSpan>>>,
216}

218/// Trait for tick closures — abstracts over both concrete async closures (wrapped in
219/// [`TickClosureFn`]) and type-erased boxed versions ([`TickClosureErased`]).
220///
221/// The `&mut Context` parameter is owned by [`Dfir`] and lent to the
222/// closure each tick, avoiding shared-ownership overhead for the context.
223///
224/// The [`Self::Metrics`] associated type carries the [`MetricsState`] typestate of the context the
225/// closure expects, so that [`Dfir<Tick>`] can be parameterized by the closure alone.
226pub trait TickClosure {
227    /// Whether the context passed to this closure tracks metrics. See [`MetricsState`].
228    type Metrics: MetricsState;
229
230    /// Call the tick closure. Returns `true` if any subgraph received input data.
231    #[doc(hidden)]
232    fn call_tick<'a>(
233        &'a mut self,
234        ctx: &'a mut Context<Self::Metrics>,
235    ) -> impl Future<Output = bool> + 'a;
236}
237
238/// Wraps an `AsyncFnMut(&mut Context<M>) -> bool` closure as a [`TickClosure`].
239///
240/// A blanket `impl<M, F: AsyncFnMut(&mut Context<M>)> TickClosure for F` is not allowed
241/// since `M` would only be constrained via the `AsyncFnMut` bound (E0207), so the generated code
242/// wraps the tick closure in this newtype which carries `M` explicitly.
243#[doc(hidden)]
244pub struct TickClosureFn<M, F> {
245    func: F,
246    _metrics: PhantomData<fn() -> M>,
247}
248
249impl<M, F> TickClosureFn<M, F>
250where
251    M: MetricsState,
252    F: for<'a> AsyncFnMut(&'a mut Context<M>) -> bool,
253{
254    /// Wrap the given async tick closure.
255    pub fn new(func: F) -> Self {
256        Self {
257            func,
258            _metrics: PhantomData,
259        }
260    }
261}
262
263impl<M, F> TickClosure for TickClosureFn<M, F>
264where
265    M: MetricsState,
266    F: for<'a> AsyncFnMut(&'a mut Context<M>) -> bool,
267{
268    type Metrics = M;
269
270    fn call_tick<'a>(&'a mut self, ctx: &'a mut Context<M>) -> impl Future<Output = bool> + 'a {
271        (self.func)(ctx)
272    }
273}

295/// Object-safe inner trait for [`TickClosureErased`]. Needed because [`TickClosure::call_tick`]
296/// returns `impl Future` (not object-safe), but a trait with `&mut self -> Pin<Box<dyn Future + '_>>`
297/// is — the returned future borrows from the trait object which owns the closure.
298trait TickClosureErasedInner<M: MetricsState> {
299    fn call_tick<'a>(
300        &'a mut self,
301        ctx: &'a mut Context<M>,
302    ) -> Pin<Box<dyn Future<Output = bool> + 'a>>;
303}
304
305impl<Tick: TickClosure> TickClosureErasedInner<Tick::Metrics> for Tick {
306    fn call_tick<'a>(
307        &'a mut self,
308        ctx: &'a mut Context<Tick::Metrics>,
309    ) -> Pin<Box<dyn Future<Output = bool> + 'a>> {
310        Box::pin(TickClosure::call_tick(self, ctx))
311    }
312}

314impl<M: MetricsState> TickClosure for TickClosureErased<M> {

317    fn call_tick<'a>(&'a mut self, ctx: &'a mut Context<M>) -> impl Future<Output = bool> + 'a {
318        self.0.call_tick(ctx)

330    /// Create a new `Dfir` from a tick closure, inline context,
331    /// and meta graph / diagnostics JSON strings.
332    #[doc(hidden)]
333    pub fn new(
334        tick_closure: Tick,
335        context: Context<Tick::Metrics>,
336        meta_graph_json: Option<&str>,
337        diagnostics_json: Option<&str>,
338    ) -> Self {
339        #[cfg(not(feature = "meta"))]
340        let _ = (meta_graph_json, diagnostics_json);
341        Self {
342            tick_closure,
343            wake_state: context.wake_state.clone(),

353                    "Expected no diagnostics, got: {:#?}",
354                    op_inst_diagnostics

356                meta_graph
357            }),
358            #[cfg(feature = "meta")]
359            diagnostics: diagnostics_json.map(|json| {
360                serde_json::from_str(json).expect("Failed to deserialize diagnostics.")
361            }),
362        }
363    }

389    /// Only available when metrics tracking is enabled, see [`MetricsState`].

391        Rc::clone(self.context.metrics())

397    /// The first call to `take_interval` returns metrics since this DFIR instance was created. Each subsequent call to
398    /// `take_interval` returns metrics since the previous call.

405        DfirMetricsIntervals {
406            curr: self.metrics(),
407            prev: None,
408        }
409    }

412impl<Tick: TickClosure> Dfir<Tick> {
413    /// Spawns all tasks buffered via [`Context::request_task`].
414    ///
415    /// This drains the buffer, so subsequent calls are no-ops until new tasks are requested.
416    #[cfg(feature = "tokio")]
417    fn spawn_tasks(&mut self) {
418        for task in self.context.tasks_to_spawn.drain(..) {
419            tokio::task::spawn_local(task);
420        }
421    }
422
423    /// Run a single tick. Returns `true` if any subgraph received input data.
424    ///
425    /// Checks both handoff buffers (via `work_done` flag set in generated recv port code)
426    /// and external events (via `can_start_tick` set by `waker`/`schedule_subgraph`).
427    pub async fn run_tick(&mut self) -> bool {
428        #[cfg(feature = "tokio")]
429        self.spawn_tasks();
430        let had_external = self
431            .wake_state
432            .can_start_tick
433            .swap(false, Ordering::Relaxed);
434        let tick_had_work = self.tick_closure.call_tick(&mut self.context).await;
435        had_external || tick_had_work || self.wake_state.can_start_tick.load(Ordering::Relaxed)
436    }
437
438    /// Run a single tick synchronously. Panics if the tick yields (async suspension).
439    /// Returns `true` if work was done (see [`Self::run_tick`]).
440    pub fn run_tick_sync(&mut self) -> bool {
441        let mut fut = std::pin::pin!(self.run_tick());
442        let mut ctx = std::task::Context::from_waker(std::task::Waker::noop());
443        match fut.as_mut().poll(&mut ctx) {
444            std::task::Poll::Ready(result) => result,
445            std::task::Poll::Pending => {
446                panic!("Dfir::run_tick_sync: tick yielded asynchronously.")
447            }
448        }
449    }
450
451    /// Run ticks as long as work is available, then return.
452    #[cfg(feature = "tokio")]
453    pub async fn run_available(&mut self) {
454        // Always run at least one tick.
455        self.wake_state
456            .can_start_tick
457            .store(false, Ordering::Relaxed);
458        loop {
459            self.run_tick().await;
460            let can_start_tick = self
461                .wake_state
462                .can_start_tick
463                .swap(false, Ordering::Relaxed);
464            if !can_start_tick {
465                break;
466            }
467            // Yield between each tick to receive more events.
468            tokio::task::yield_now().await;
469        }
470    }
471
472    /// [`Self::run_available`] but panics if any tick yields asynchronously.
473    pub fn run_available_sync(&mut self) {
474        self.wake_state
475            .can_start_tick
476            .store(false, Ordering::Relaxed);
477        loop {
478            self.run_tick_sync();
479            let can_start_tick = self
480                .wake_state
481                .can_start_tick
482                .swap(false, Ordering::Relaxed);
483            if !can_start_tick {
484                break;
485            }
486        }
487    }
488
489    /// Run forever, processing ticks when work is available and yielding when idle.
490    #[cfg(feature = "tokio")]
491    pub async fn run(&mut self) -> crate::Never {
492        loop {
493            self.run_available().await;
494            // Wait for an external event to wake us.
495            std::future::poll_fn(|cx| {
496                // Register waker first to avoid race: if an event fires between
497                // the check and the register, the waker is already in place.
498                self.wake_state.task_waker.register(cx.waker());
499                if self.wake_state.can_start_tick.load(Ordering::Relaxed) {
500                    std::task::Poll::Ready(())
501                } else {
502                    std::task::Poll::Pending
503                }
504            })
505            .await;
506        }
507    }
508}

510impl<Tick: TickClosure + 'static> Dfir<Tick> {
511    /// Type-erase the tick closure for use in heterogeneous collections.
512    ///
513    /// Wraps the concrete tick closure in [`TickClosureErased`], which boxes the future
514    /// returned by each tick call. This adds one heap allocation per tick, but enables
515    /// storing multiple `Dfir`s with different closure types in a single `Vec`.
516    ///
517    /// The [`MetricsState`] typestate is preserved: the result is `DfirErased<Tick::Metrics>`.
518    ///
519    /// Only needed for the sim runtime path. The trybuild and embedded paths keep the
520    /// concrete type and pay no erasure cost.
521    pub fn into_erased(self) -> DfirErased<Tick::Metrics> {
522        Dfir {
523            tick_closure: TickClosureErased(Box::new(self.tick_closure)),
524            wake_state: self.wake_state,
525            context: self.context,
526            #[cfg(feature = "meta")]
527            meta_graph: self.meta_graph,
528            #[cfg(feature = "meta")]
529            diagnostics: self.diagnostics,
530        }
531    }
