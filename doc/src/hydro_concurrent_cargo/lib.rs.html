19/// Log a timestamped message to the build coordination log.
20pub fn log_build_event(project_dir: &Path, msg: &str) {
21    let log_path = project_dir.join("build-coordination.log");
22    let mut file = fs::OpenOptions::new()
23        .create(true)
24        .append(true)
25        .open(log_path)
26        .unwrap();
27    let now = std::time::SystemTime::now()
28        .duration_since(std::time::UNIX_EPOCH)
29        .unwrap()
30        .as_millis();
31    let thread_id = std::thread::current().id();
32    let _ = writeln!(file, "[{now}ms] [{thread_id:?}] {msg}");
33}

39/// Global prebuild serialization lock (in-process).
40static GLOBAL_PREBUILD_LOCK: parking_lot::RwLock<()> = parking_lot::RwLock::new(());
41
42/// Per-key in-process prebuild locks, mirroring the per-key lock files (see [`run_prebuild`]).
43static DEP_BUILD_LOCKS_PER_KEY: LazyLock<Mutex<HashMap<String, &'static parking_lot::RwLock<()>>>> =
44    LazyLock::new(|| Mutex::new(HashMap::new()));
45
46fn get_dep_lock(key: &str) -> &'static parking_lot::RwLock<()> {
47    let mut map = DEP_BUILD_LOCKS_PER_KEY.lock().unwrap();
48    map.entry(key.to_owned())
49        .or_insert_with(|| Box::leak(Box::new(parking_lot::RwLock::new(()))))
50}
51
52/// Per-job-dir mutexes for serializing job dir setup/population within a process.
53static JOB_DIR_LOCKS: LazyLock<Mutex<HashMap<PathBuf, &'static Mutex<()>>>> =
54    LazyLock::new(|| Mutex::new(HashMap::new()));
55
56fn get_job_dir_lock(job_dir: &Path) -> &'static Mutex<()> {
57    let mut map = JOB_DIR_LOCKS.lock().unwrap();
58    map.entry(job_dir.to_owned())
59        .or_insert_with(|| Box::leak(Box::new(Mutex::new(()))))
60}

66/// Guard holding the cargo build directory file lock (shared).
67pub struct CargoBuildLock {
68    _file: fs::File,
69}
70
71impl CargoBuildLock {
72    pub fn lock_shared(lock_path: &Path) -> Self {
73        let file = fs::OpenOptions::new()
74            .read(true)
75            .write(true)
76            .create(true)
77            .truncate(false)
78            .open(lock_path)
79            .unwrap();
80        file.lock_shared().unwrap();
81        CargoBuildLock { _file: file }
82    }
83}

89/// Guard holding both in-process and file locks for prebuild coordination.
90pub struct PrebuildGuard {
91    _rw_guard: RwGuard,
92    _global_guard: Option<parking_lot::RwLockWriteGuard<'static, ()>>,
93    _global_file: Option<fs::File>,
94    _file: fs::File,
95    lock_dir: PathBuf,
96}
97
98#[expect(
99    dead_code,
100    reason = "variants hold lock guards that are released on drop"
101)]
102enum RwGuard {
103    Read(parking_lot::RwLockReadGuard<'static, ()>),
104    Upgradable(parking_lot::RwLockUpgradableReadGuard<'static, ()>),
105    Write(parking_lot::RwLockWriteGuard<'static, ()>),
106}
107
108impl PrebuildGuard {
109    /// Acquire an upgradable lock (in-process upgradable read + file shared). `key` selects
110    /// the in-process lock and should identify `lock_path`.
111    pub fn lock_upgradable(lock_path: &Path, key: &str) -> Self {
112        let rw_guard = get_dep_lock(key).upgradable_read();
113        let lock_dir = lock_path.parent().unwrap().to_owned();
114        let file = fs::OpenOptions::new()
115            .read(true)
116            .write(true)
117            .create(true)
118            .truncate(false)
119            .open(lock_path)
120            .unwrap();
121        file.lock_shared().unwrap();
122        PrebuildGuard {
123            _rw_guard: RwGuard::Upgradable(rw_guard),
124            _global_guard: None,
125            _global_file: None,
126            _file: file,
127            lock_dir,
128        }
129    }
130
131    /// Upgrade to exclusive (in-process write + file exclusive + global locks).
132    pub fn upgrade(self) -> Self {
133        let file = self._file;
134        let rw_guard = match self._rw_guard {
135            RwGuard::Upgradable(u) => parking_lot::RwLockUpgradableReadGuard::upgrade(u),
136            _ => panic!("can only upgrade from upgradable"),
137        };
138        // Release per-key shared lock before acquiring global exclusive
139        // to avoid deadlock (other processes hold per-key shared and wait for global).
140        file.unlock().unwrap();
141        let global_guard = GLOBAL_PREBUILD_LOCK.write();
142        let global_file = fs::OpenOptions::new()
143            .read(true)
144            .write(true)
145            .create(true)
146            .truncate(false)
147            .open(self.lock_dir.join(".global-prebuild.lock"))
148            .unwrap();
149        global_file.lock().unwrap();
150        file.lock().unwrap();
151        PrebuildGuard {
152            _rw_guard: RwGuard::Write(rw_guard),
153            _global_guard: Some(global_guard),
154            _global_file: Some(global_file),
155            _file: file,
156            lock_dir: self.lock_dir,
157        }
158    }
159
160    /// Downgrade from exclusive to shared (releases global lock).
161    pub fn downgrade(self) -> Self {
162        let file = self._file;
163        file.lock_shared().unwrap();
164        let rw_guard = match self._rw_guard {
165            RwGuard::Write(w) => RwGuard::Read(parking_lot::RwLockWriteGuard::downgrade(w)),
166            RwGuard::Upgradable(u) => {
167                RwGuard::Read(parking_lot::RwLockUpgradableReadGuard::downgrade(u))
168            }
169            r @ RwGuard::Read(_) => r,
170        };
171        PrebuildGuard {
172            _rw_guard: rw_guard,
173            _global_guard: None,
174            _global_file: None,
175            _file: file,
176            lock_dir: self.lock_dir,
177        }
178    }
179
180    /// Get mutable access to the underlying file (for writing timestamps).
181    pub fn file_mut(&mut self) -> &mut fs::File {
182        &mut self._file
183    }
184}

190/// Set up a job directory with symlinked .fingerprint and deps subdirs.
191/// Does NOT set up build/ — use `populate_job_build_dir` for final builds
192/// or manually symlink for prebuild.
193pub fn setup_job_dir(jobs_dir: &Path, name: &str, shared_debug: &Path) -> PathBuf {
194    let job = jobs_dir.join(name);
195    let _lock = get_job_dir_lock(&job);
196    let _lock = _lock.lock().unwrap();
197
198    // File lock for cross-process serialization on the same job dir.
199    fs::create_dir_all(&job).unwrap();
200    let job_lock_file = fs::OpenOptions::new()
201        .read(true)
202        .write(true)
203        .create(true)
204        .truncate(false)
205        .open(job.join(".job.lock"))
206        .unwrap();
207    job_lock_file.lock().unwrap();
208
209    let job_debug = job.join("debug");
210    fs::create_dir_all(&job_debug).unwrap();
211    for subdir in [".fingerprint", "deps"] {
212        let link = job_debug.join(subdir);
213        if !link.exists() {
214            let target = shared_debug.join(subdir);
215            fs::create_dir_all(&target).unwrap();
216            #[cfg(unix)]
217            std::os::unix::fs::symlink(&target, &link).unwrap();
218            #[cfg(windows)]
219            std::os::windows::fs::symlink_dir(&target, &link).unwrap();
220        }
221    }
222    job
223}

225/// Symlink the build/ directory for prebuild jobs (writes through to shared).
226pub fn symlink_prebuild_build_dir(prebuild_target: &Path, shared_debug: &Path) {
227    let link = prebuild_target.join("debug").join("build");
228    if !link.exists() {
229        let target = shared_debug.join("build");
230        fs::create_dir_all(&target).unwrap();
231        #[cfg(unix)]
232        std::os::unix::fs::symlink(&target, &link).unwrap();
233        #[cfg(windows)]
234        std::os::windows::fs::symlink_dir(&target, &link).unwrap();
235    }
236}

241    _file: fs::File,

244/// Populate the per-job build/ directory from the shared build/ directory.
245/// Returns a guard that holds the job lock — keep alive for the entire final build.
246/// This prevents races from cargo's `link_or_copy` on build script binaries.
247pub fn populate_job_build_dir(job_debug: &Path, shared_debug: &Path) -> JobBuildGuard {
248    let shared_build = shared_debug.join("build");
249    let job_build = job_debug.join("build");
250
251    let job_dir = job_debug.parent().unwrap();
252    let mutex_guard = get_job_dir_lock(job_dir);
253    let mutex_guard = mutex_guard.lock().unwrap();
254
255    // Also hold a file lock for cross-process serialization on the same job dir.
256    let job_lock_path = job_dir.join(".job.lock");
257    let job_lock_file = fs::OpenOptions::new()
258        .read(true)
259        .write(true)
260        .create(true)
261        .truncate(false)
262        .open(&job_lock_path)
263        .unwrap();
264    job_lock_file.lock().unwrap();
265
266    let _ = fs::remove_dir_all(&job_build);
267    fs::create_dir_all(&job_build).unwrap();
268    if shared_build.exists() {
269        for entry in fs::read_dir(&shared_build).unwrap() {
270            let entry = entry.unwrap();
271            if !entry.file_type().unwrap().is_dir() {
272                continue;
273            }
274            let dest = job_build.join(entry.file_name());
275            // Create dir and symlink each child individually so cargo can
276            // safely remove+relink build-script-build without racing other jobs.
277            fs::create_dir_all(&dest).unwrap();
278            for file in fs::read_dir(entry.path()).unwrap() {
279                let file = file.unwrap();
280                let file_dest = dest.join(file.file_name());
281                #[cfg(unix)]
282                std::os::unix::fs::symlink(file.path(), &file_dest).unwrap();
283                #[cfg(windows)]
284                {
285                    if file.file_type().unwrap().is_dir() {
286                        std::os::windows::fs::symlink_dir(file.path(), &file_dest).unwrap();
287                    } else {
288                        std::os::windows::fs::symlink_file(file.path(), &file_dest).unwrap();
289                    }
290                }
291            }
292        }
293    }
294
295    JobBuildGuard {
296        _mutex_guard: mutex_guard,
297        _file: job_lock_file,
298    }
299}

305/// The verbose version (`rustc -vV`) of the compiler child `cargo` invocations will use,
306/// honoring the `RUSTC` override like cargo does.
307static RUSTC_VERBOSE_VERSION: LazyLock<String> = LazyLock::new(|| {
308    let rustc = std::env::var_os("RUSTC").unwrap_or_else(|| "rustc".into());
309    let output = std::process::Command::new(rustc)
310        .arg("-vV")
311        .output()
312        .expect("failed to run `rustc -vV`");
313    assert!(output.status.success(), "`rustc -vV` failed");
314    String::from_utf8(output.stdout).unwrap()
315});
316
317fn hash_hex(hash: impl FnOnce(&mut DefaultHasher)) -> String {
318    let mut hasher = DefaultHasher::new();
319    hash(&mut hasher);
320    format!("{:016x}", hasher.finish())
321}
322
323/// Fingerprint of the compilation *environment* a child build runs under: the rustflags (in
324/// any stable encoding) and the compiler's verbose version.
325///
326/// Cargo keys the filenames of rlibs on both of these, so different fingerprints never
327/// collide in the shared `deps/` and can be cached side by side. It strips that key from
328/// dylib filenames unless `__CARGO_DEFAULT_LIB_METADATA` is set, so without that variable the
329/// trybuild dylib bakes this fingerprint into its target name instead (see [`dylib_lib_name`]).
330pub fn toolchain_fingerprint(rustflags: &str) -> String {
331    hash_hex(|h| {
332        rustflags.hash(h);
333        RUSTC_VERBOSE_VERSION.hash(h);
334    })
335}
336
337/// Whether cargo hashes the filenames of path-package dylibs. Cargo normally keeps those
338/// stable (`libfoo.so`) so executables can embed them; the undocumented
339/// `__CARGO_DEFAULT_LIB_METADATA` variable — which rust-lang/rust's bootstrap uses to build
340/// libstd — forces the metadata hash onto them too, making every configuration's dylib a
341/// distinct file.
342fn lib_metadata_enabled() -> bool {
343    std::env::var_os("__CARGO_DEFAULT_LIB_METADATA").is_some()
344}
345
346/// The `[lib] name` of the trybuild dylib package `package_name` when built under `rustflags`.
347///
348/// With `__CARGO_DEFAULT_LIB_METADATA` set this is cargo's default (the package name with
349/// `-` replaced by `_`): cargo already hashes the filename per configuration, so the manifest
350/// never needs to change. Without it the [`toolchain_fingerprint`] is appended so that
351/// configurations get distinct `lib{name}.so` files and cache side by side instead of
352/// clobbering one another. Features are not part of the name: they vary per build against one
353/// shared manifest, and the prebuild lock already serializes feature switches.
354pub fn dylib_lib_name(package_name: &str, rustflags: &str) -> String {
355    let default = package_name.replace('-', "_");
356    if lib_metadata_enabled() {
357        default
358    } else {
359        format!("{default}_{}", toolchain_fingerprint(rustflags))
360    }
361}

363/// Exclusive file lock on a trybuild project's `.hydro-trybuild-lock`, serializing all writes
364/// to the project's generated files (manifests, sources) across processes. Released on drop.
365pub fn lock_project(project_dir: &Path) -> fs::File {
366    let file = fs::File::create(project_dir.join(".hydro-trybuild-lock")).unwrap();
367    file.lock().unwrap();
368    file
369}

371fn dylib_manifest_path(project_dir: &Path) -> PathBuf {
372    project_dir.join("dylib").join("Cargo.toml")
373}
374
375fn read_manifest(path: &Path) -> Option<toml_edit::DocumentMut> {
376    fs::read_to_string(path).ok()?.parse().ok()
377}
378
379fn manifest_lib_name(doc: &toml_edit::DocumentMut) -> Option<&str> {
380    doc.get("lib")?.get("name")?.as_str()
381}
382
383/// The `[lib] name` currently in the dylib manifest of the trybuild project at `project_dir`,
384/// if it has one. Call with [`lock_project`] held and write the manifest under the same lock.
385///
386/// Manifest generation must reuse this rather than compute a fresh name: the name only ever
387/// changes under the exclusive prebuild lock (see [`run_prebuild`]), which is what guarantees
388/// no in-flight final build links against a dylib named for another configuration.
389pub fn current_dylib_lib_name(project_dir: &Path) -> Option<String> {
390    let doc = read_manifest(&dylib_manifest_path(project_dir))?;
391    manifest_lib_name(&doc).map(str::to_owned)
392}
393
394/// Point the project's dylib manifest at the [`dylib_lib_name`] for `rustflags`.
395///
396/// Dylib builds call this first thing in their [`run_prebuild`] `build_fn`, i.e. under the
397/// exclusive prebuild lock, once no in-flight final build links the old name. It takes
398/// [`lock_project`] so a concurrent manifest regeneration cannot read the old name and write it
399/// back over the new one.
400pub fn set_dylib_lib_name(project_dir: &Path, rustflags: &str) {
401    let _project_lock = lock_project(project_dir);
402    let manifest = dylib_manifest_path(project_dir);
403    let mut doc = read_manifest(&manifest).expect("dylib manifest must exist before prebuild");
404    let package_name = doc["package"]["name"].as_str().unwrap();
405    let lib_name = dylib_lib_name(package_name, rustflags);
406    if manifest_lib_name(&doc) != Some(&lib_name) {
407        doc["lib"]["name"] = toml_edit::value(lib_name);
408        fs::write(&manifest, doc.to_string()).unwrap();
409    }
410}

412/// Run the prebuild phase with proper locking and freshness checking.
413///
414/// - `target_dir`: the shared target directory (e.g. `target/`)
415/// - `crate_name`: unique identifier for the crate being built (included in hash)
416/// - `features`: list of features for hashing
417/// - `rustflags`: the rustflags the prebuild (and the final builds sharing its artifacts) are
418///   compiled with; see [`toolchain_fingerprint`]
419/// - `staged_paths`: paths to check mtime against for freshness
420/// - `build_fn`: closure called with `prebuild_target` path under the exclusive lock; should
421///   run cargo build(s), and for dylib projects first call [`set_dylib_lib_name`]
422///
423/// Returns `(PrebuildGuard, CargoBuildLock)` both held in shared mode.
424/// Caller should keep both alive during the final build.
425///
426/// # Locking and freshness
427///
428/// Cargo itself supports one build per target dir; the per-job target dirs sidestep that, so
429/// concurrent builds are safe only while everything they both write has a distinct filename.
430/// Cargo hashes rustflags, compiler version and features into every filename except a path
431/// package's dylib (unless `__CARGO_DEFAULT_LIB_METADATA` is set), so the dylib and the
432/// manifest naming it are the one shared resource:
433///
434/// - Without `__CARGO_DEFAULT_LIB_METADATA` there is a single `.prebuild.lock`. Every build
435///   holds it shared through its final build; switching configuration takes it exclusive,
436///   which waits for all in-flight builds to finish before `build_fn` renames the manifest and
437///   rebuilds the prebuild. A fresh stamp therefore implies the manifest names this
438///   configuration's dylib.
439/// - With it, cargo names every configuration's files distinctly and the manifest is constant,
440///   so builds only need to serialize with others of the same `(crate, features, fingerprint)`
441///   — one `.prebuild-<key>.lock` each — and configurations never wait on one another.
442///
443/// Running builds with and without the variable against one target dir is unsupported: they
444/// would rename the shared manifest under different locks.
445///
446/// The freshness stamp stored in the lock file covers crate, features and fingerprint, so a
447/// prebuild for a different configuration is never mistaken for fresh.
448pub fn run_prebuild(
449    target_dir: &Path,
450    crate_name: &str,
451    features: &[String],
452    rustflags: &str,
453    staged_paths: &[PathBuf],
454    build_fn: impl FnOnce(&Path),
455) -> (PrebuildGuard, CargoBuildLock) {
456    // Acquire cargo build lock shared for the entire prebuild + final build duration.
457    let shared_debug = target_dir.join("debug");
458    fs::create_dir_all(&shared_debug).ok();
459    let cargo_lock = CargoBuildLock::lock_shared(&shared_debug.join(".cargo-build-lock"));
460
461    let toolchain = toolchain_fingerprint(rustflags);
462    let fingerprint = hash_hex(|h| {
463        crate_name.hash(h);
464        let mut sorted = features.to_vec();
465        sorted.sort();
466        sorted.hash(h);
467        toolchain.hash(h);
468    });
469
470    // Without `__CARGO_DEFAULT_LIB_METADATA` every configuration shares one fixed key, so this
471    // lock doubles as the global one: a configuration switch (exclusive) drains every in-flight
472    // build, not just those of the same configuration. `.global-prebuild.lock` (taken in
473    // `upgrade`) only adds anything in the per-key case, where it keeps two configurations from
474    // running their `cargo build`s into the shared `deps/` at the same time.
475    let lock_key = if lib_metadata_enabled() {
476        format!("prebuild-{fingerprint}")
477    } else {
478        "prebuild".to_owned()
479    };
480    let lock_path = target_dir.join(format!(".{lock_key}.lock"));
481
482    let staged_mtime = staged_paths
483        .iter()
484        .filter_map(|p| fs::metadata(p).and_then(|m| m.modified()).ok())
485        .max()
486        .unwrap_or(SystemTime::UNIX_EPOCH);
487
488    let prebuild_is_fresh = |path: &Path, expected_hash: &str| -> bool {
489        fs::read_to_string(path)
490            .ok()
491            .and_then(|s| {
492                let mut parts = s.trim().splitn(2, ':');
493                let hash = parts.next()?;
494                let nanos = parts.next()?.parse::<u128>().ok()?;
495                let ts = SystemTime::UNIX_EPOCH
496                    .checked_add(std::time::Duration::from_nanos(nanos as u64))?;
497                Some((hash.to_owned(), ts))
498            })
499            .is_some_and(|(hash, ts)| hash == expected_hash && ts >= staged_mtime)
500    };
501
502    log_build_event(
503        target_dir,
504        &format!("prebuild: lock_upgradable, lock_key={lock_key}, fingerprint={fingerprint}"),
505    );
506    let guard = PrebuildGuard::lock_upgradable(&lock_path, &lock_key);
507    if prebuild_is_fresh(&lock_path, &fingerprint) {
508        log_build_event(target_dir, "prebuild: fresh, downgrading to shared");
509        return (guard.downgrade(), cargo_lock);
510    }
511
512    log_build_event(target_dir, "prebuild: not fresh, upgrading to exclusive");
513    let mut guard = guard.upgrade();
514    log_build_event(target_dir, "prebuild: exclusive acquired");
515
516    // Re-check after acquiring exclusive.
517    if !prebuild_is_fresh(&lock_path, &fingerprint) {
518        let shared_debug = target_dir.join("debug");
519        let jobs_dir = target_dir.join("jobs");
520        let prebuild_target = setup_job_dir(&jobs_dir, "prebuild", &shared_debug);
521        symlink_prebuild_build_dir(&prebuild_target, &shared_debug);
522
523        build_fn(&prebuild_target);
524
525        // Write fingerprint:timestamp.
526        use std::io::Seek;
527        let now_nanos = SystemTime::now()
528            .duration_since(SystemTime::UNIX_EPOCH)
529            .unwrap()
530            .as_nanos();
531        let file = guard.file_mut();
532        file.set_len(0).unwrap();
533        file.seek(std::io::SeekFrom::Start(0)).unwrap();
534        write!(file, "{}:{}", fingerprint, now_nanos).unwrap();
535    }
536
537    (guard.downgrade(), cargo_lock)
538}
