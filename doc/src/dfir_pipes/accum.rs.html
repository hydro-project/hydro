1//! Shared accumulator state machinery used by both the pull-based and push-based `Accumulate`
2//! combinators ([`crate::pull::Accumulate`] and [`crate::push::Accumulate`]).
3//!
4//! The [`AccumState`] trait defines how items are accumulated and how the accumulated state is
5//! drained into an iterator on finalization. Each struct here encapsulates the accumulation
6//! logic and drain behavior for a specific operator (fold, reduce, sort), in both owned and
7//! borrowed modes.
8
9use core::borrow::BorrowMut;
10use core::iter::Once;
11use core::marker::PhantomData;
12
13#[cfg(any(test, feature = "alloc"))]
14extern crate alloc;
15
16/// Trait for accumulator state used by the accumulate combinators
17/// ([`crate::pull::Accumulate`] and [`crate::push::Accumulate`]).
18///
19/// Implementors define how items are accumulated and how the accumulated state
20/// is drained into an iterator on finalization.
21///
22/// The key insight is that [`AccumState::into_iter`] consumes `self` by value,
23/// which allows borrowed-mode implementations to return references with the
24/// original lifetime (e.g., `&'a mut T` from `&'a mut Option<T>`).
25///
26/// # Two modes
27///
28/// - **Owned / `'tick` mode**: The state is created fresh each tick and consumed
29///   on finalize. Example: `FoldState<i32, F, i32, Item>` yields `Once<i32>`.
30/// - **Borrowed / `'static` mode**: The state borrows externally-owned storage
31///   that persists across ticks. Example: `FoldState<&'a mut i32, F, i32, Item>`
32///   yields `Once<&'a mut i32>`.
33pub trait AccumState: Sized {
34    /// The type of items being accumulated (input to the combinator).
35    type Input;
36    /// The type of items emitted downstream after finalization.
37    type Output;
38    /// The iterator type returned by [`AccumState::into_iter`].
39    type Iter: Iterator<Item = Self::Output>;
40
41    /// Fold an incoming item into the accumulator.
42    fn accumulate(&mut self, item: Self::Input);
43
44    /// Consume the accumulator and return an iterator over the output items.
45    ///
46    /// This is called once during finalization. Because it takes `self` by
47    /// value, borrowed-mode implementations can release their full lifetime.
48    fn into_iter(self) -> Self::Iter;
49
50    /// Returns the size hint for the output iterator, given the input's size hint.
51    ///
52    /// This is called during `size_hint` while still accumulating, allowing the
53    /// combinator to inform downstream of expected output cardinality before
54    /// finalization occurs.
55    ///
56    /// The default returns `(0, None)` (unknown).
57    fn size_hint(&self, _input_hint: (usize, Option<usize>)) -> (usize, Option<usize>) {
58        (0, None)
59    }
60}

62// ============================================================================
63// Fold (unified: owned T or borrowed &'a mut T via BorrowMut)
64// ============================================================================
65
66/// Accumulator state for fold, supporting both owned and borrowed modes
67/// via [`BorrowMut`].
68///
69/// - **Owned mode** (`Accum = T`): accumulates into an owned `T`, emits `T` downstream.
70/// - **Borrowed mode** (`Accum = &'a mut T`): accumulates into a reference,
71///   emits `&'a mut T` downstream. The value persists across ticks.
72///
73/// `AccumInner` is the type you actually fold into (what `BorrowMut` resolves to).
74pub struct FoldState<Accum, F, AccumInner, Item> {
75    /// The accumulator — either an owned value or a mutable reference.
76    pub accum: Accum,
77    /// The combining function: `(&mut AccumInner, Item) -> ()`.
78    pub comb_fn: F,
79    /// Marker for `AccumInner` and `Item`.
80    pub _phantom: PhantomData<fn(&mut AccumInner, Item)>,
81}
82
83impl<Accum, F, AccumInner, Item> FoldState<Accum, F, AccumInner, Item> {
84    /// Creates a new `FoldState` with the given accumulator and combining function.
85    pub const fn new(accum: Accum, comb_fn: F) -> Self {
86        Self {
87            accum,
88            comb_fn,
89            _phantom: PhantomData,
90        }
91    }
92}
93
94impl<Accum, F, AccumInner, Item> AccumState for FoldState<Accum, F, AccumInner, Item>
95where
96    Accum: BorrowMut<AccumInner>,
97    F: FnMut(&mut AccumInner, Item),
98{
99    type Input = Item;
100    type Output = Accum;
101    type Iter = Once<Accum>;
102
103    fn accumulate(&mut self, item: Item) {
104        (self.comb_fn)(self.accum.borrow_mut(), item);
105    }
106
107    fn into_iter(self) -> Self::Iter {
108        core::iter::once(self.accum)
109    }
110
111    fn size_hint(&self, _input_hint: (usize, Option<usize>)) -> (usize, Option<usize>) {
112        (1, Some(1))
113    }
114}

116// ============================================================================
117// Reduce (owned mode: Option<T>, borrowed mode: &'a mut Option<T>)
118// ============================================================================
119
120/// Accumulator state for reduce.
121///
122/// - **Owned mode** (`Accum = Option<T>`): the first item initializes the
123///   accumulator, subsequent items are merged. Takes the value on finalize,
124///   emitting 0 or 1 `T` downstream.
125/// - **Borrowed mode** (`Accum = &'a mut Option<T>`): same accumulation logic,
126///   but the option persists across ticks. Emits `&'a mut T` on finalize
127///   (or nothing if no items were received).
128pub struct ReduceState<Accum, F, T> {
129    /// The accumulator — either `Option<T>` or `&'a mut Option<T>`.
130    pub accum: Accum,
131    /// The reduce function: `(&mut T, T) -> ()`.
132    pub reduce_fn: F,
133    /// Marker for the item type.
134    pub _phantom: PhantomData<fn(&mut T, T)>,
135}
136
137impl<Accum, F, T> ReduceState<Accum, F, T> {
138    /// Creates a new `ReduceState` with the given accumulator and reduce function.
139    pub const fn new(accum: Accum, reduce_fn: F) -> Self {
140        Self {
141            accum,
142            reduce_fn,
143            _phantom: PhantomData,
144        }
145    }
146}
147
148/// Owned mode: accumulates into `Option<T>`, takes value on finalize.
149impl<F, T> AccumState for ReduceState<Option<T>, F, T>
150where
151    F: FnMut(&mut T, T),
152{
153    type Input = T;
154    type Output = T;
155    type Iter = core::option::IntoIter<T>;
156
157    fn accumulate(&mut self, item: T) {
158        match &mut self.accum {
159            Some(acc) => (self.reduce_fn)(acc, item),
160            None => self.accum = Some(item),
161        }
162    }
163
164    fn into_iter(self) -> Self::Iter {
165        self.accum.into_iter()
166    }
167
168    fn size_hint(&self, _input_hint: (usize, Option<usize>)) -> (usize, Option<usize>) {
169        (0, Some(1))
170    }
171}

173/// Borrowed mode: accumulates into `&'a mut Option<T>`, value persists across ticks.
174/// Emits `&'a mut T` on finalize (nothing if empty).
175impl<'a, F, T> AccumState for ReduceState<&'a mut Option<T>, F, T>
176where
177    F: FnMut(&mut T, T),
178{
179    type Input = T;
180    type Output = &'a mut T;
181    type Iter = core::option::IntoIter<&'a mut T>;
182
183    fn accumulate(&mut self, item: T) {
184        match self.accum {
185            Some(acc) => (self.reduce_fn)(acc, item),
186            None => *self.accum = Some(item),
187        }
188    }
189
190    fn into_iter(self) -> Self::Iter {
191        self.accum.as_mut().into_iter()
192    }
193
194    fn size_hint(&self, _input_hint: (usize, Option<usize>)) -> (usize, Option<usize>) {
195        (0, Some(1))
196    }
197}

199// ============================================================================
200// Sort (owned mode, requires alloc)
201// ============================================================================
202
203/// Accumulator state for sort.
204///
205/// Collects all items into a `Vec`, sorts them on finalize, and emits
206/// them in sorted order.
207#[cfg(any(test, feature = "alloc"))]
208#[cfg_attr(docsrs, doc(cfg(feature = "alloc")))]
209pub struct SortState<T> {
210    /// Buffer for collected items.
211    pub buf: alloc::vec::Vec<T>,
212}
213
214#[cfg(any(test, feature = "alloc"))]
215#[cfg_attr(docsrs, doc(cfg(feature = "alloc")))]
216impl<T> SortState<T> {
217    /// Creates a new empty `SortState`.
218    pub const fn new() -> Self {
219        Self {
220            buf: alloc::vec::Vec::new(),
221        }
222    }
223}
224
225#[cfg(any(test, feature = "alloc"))]
226impl<T> Default for SortState<T> {
227    fn default() -> Self {
228        Self::new()
229    }
230}
231
232#[cfg(any(test, feature = "alloc"))]
233#[cfg_attr(docsrs, doc(cfg(feature = "alloc")))]
234impl<T: Ord> AccumState for SortState<T> {
235    type Input = T;
236    type Output = T;
237    type Iter = alloc::vec::IntoIter<T>;
238
239    fn accumulate(&mut self, item: T) {
240        self.buf.push(item);
241    }
242
243    fn into_iter(self) -> Self::Iter {
244        let mut buf = self.buf;
245        buf.sort_unstable();
246        buf.into_iter()
247    }
248
249    fn size_hint(&self, input_hint: (usize, Option<usize>)) -> (usize, Option<usize>) {
250        // Sort preserves cardinality: output count = current buffer + remaining input.
251        let buffered = self.buf.len();
252        let lower = buffered + input_hint.0;
253        let upper = input_hint.1.map(|u| buffered + u);
254        (lower, upper)
255    }
256}
