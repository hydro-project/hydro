1//! Test-side API for **simulator hooks**: scripting the decisions of unsafe operators.
2//!
3//! A hook handle (see [`crate::sim_hooks`]) is created from the
4//! [`FlowBuilder`](crate::compile::builder::FlowBuilder) via
5//! [`FlowBuilder::sim_hook`](crate::compile::builder::FlowBuilder::sim_hook) and attached
6//! to one specific unsafe operator with `nondet!(/** reason */ hook = handle)`. Inside a
7//! simulation test body (under [`SimFlow::deterministic`](crate::sim::flow::SimFlow::deterministic),
8//! [`fuzz`](crate::sim::flow::SimFlow::fuzz), or
9//! [`exhaustive`](crate::sim::flow::SimFlow::exhaustive)), the handle scripts the
10//! operator's decisions.
11//!
12//! # The script is a schedule
13//!
14//! Decision calls are `async`, and the sequence of calls in the test body is a
15//! **schedule**, read in program order:
16//!
17//! - Consecutive decisions that target *different hooks of the same tick* form a
18//!   **group**: one execution of that tick will consume all of them together.
19//! - A decision that targets a different tick — or the *next execution* of the same tick
20//!   (scripting a hook that already has a decision in the current group) — starts a new
21//!   group. The `.await` on the first decision of a new group suspends the test until the
22//!   previous group's tick execution has actually happened, so the test body advances in
23//!   lockstep with the execution it describes. Decisions whose turn has already come
24//!   return immediately without suspending.
25//! - Output awaits are group barriers: an output await completes only after every
26//!   decision scripted so far has been consumed.
27//!
28//! A decision may be scripted before its data exists (`release(3)` immediately after
29//! `send_many([1, 2, 3])`): the tick simply fires at the first moment the decision can be
30//! honored in full. A decision that can *never* be honored is reported when the
31//! simulation runs out of other work, attributed to the test line that is suspended
32//! waiting on it.
33//!
34//! # Holding data on purpose
35//!
36//! A hook with buffered data and no decision is an error the simulator reports at every
37//! scheduling boundary. When buffering *is* the scenario, declare it with the `pause`
38//! family ([`BatchHook::pause`], [`BatchHook::pause_while`],
39//! [`BatchHook::pause_until_count`], [`BatchHook::auto_pause`], and the snapshot
40//! equivalents).
41//!
42//! # Hooks on clusters
43//!
44//! A handle bound to an operator running on a cluster controls one independent hook
45//! instance per member; its [`OnCluster`] scope requires selecting the instance to
46//! script with `.on(member_id)` (e.g. `hook.on(0).release(2).await`). Decisions for
47//! different members never share a group — each member's execution takes its own place
48//! in the schedule — and every member's instance is independently subject to the
49//! missing-decision error.

81/// A scripted decision that has been issued but not yet installed into the schedule.
82///
83/// Awaiting it suspends the test until every previously scripted tick execution the
84/// decision must come after has actually happened (see the module docs); it resolves once
85/// the decision is installed for its tick's next execution. Panics (at the `.await`'s
86/// location) if the decision can never take its place in the schedule.
87#[must_use = "a scripted decision does nothing until awaited"]
88pub struct DecisionFuture {
89    hook_id: usize,
90    /// The cluster member whose hook instance the decision targets (from `.on(..)` on
91    /// the handle); `None` for hooks on processes.
92    member: Option<u32>,
93    /// The decision, bincode-serialized (the handle and the hook it is bound to
94    /// statically know the same decision type). `None` once installed.
95    blob: Option<Vec<u8>>,
96}
97
98impl DecisionFuture {
99    fn new(hook_id: usize, member: Option<u32>, decision: &impl ScriptDecision) -> Self {
100        DecisionFuture {
101            hook_id,
102            member,
103            blob: Some(bincode::serialize(decision).unwrap()),
104        }
105    }
106}

108/// Panics (at the scripting call site) when a per-key decision names the same key more
109/// than once, establishing the no-duplicate-keys invariant of the keyed decisions before
110/// they are installed.
111#[track_caller]
112fn assert_distinct_keys<'a, K: std::hash::Hash + Eq + 'a>(
113    keys: impl Iterator<Item = &'a K>,
114    method: &str,
115) {
116    let mut seen: dfir_rs::rustc_hash::FxHashSet<&K> = Default::default();
117    for (position, key) in keys.enumerate() {
118        assert!(
119            seen.insert(key),
120            "{}: the same key appears more than once in a single decision (duplicate at entry {}); a key takes exactly one decision per tick",
121            method,
122            position
123        );
124    }
125}

127/// Panics (at the scripting call site) when a retries decision requests zero copies:
128/// releasing an element zero times would be a *loss*, which `AtLeastOnce` does not
129/// permit.
130#[track_caller]
131fn assert_copies_at_least_one(copies: usize) {
132    assert!(
133        copies >= 1,
134        "the number of copies must be at least 1 (at-least-once permits duplicates, not losses)"
135    );
136}

138impl Future for DecisionFuture {
139    type Output = ();
140
141    #[track_caller]
142    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
143        let this = self.get_mut();
144        let Some(blob) = this.blob.take() else {
145            return Poll::Ready(());
146        };
147
148        let ctx = script_ctx();
149        match ctx.try_schedule_decision(this.hook_id, this.member, blob) {
150            Ok(ScheduleDecision::Installed) => Poll::Ready(()),
151            Ok(ScheduleDecision::Wait(blob)) => {
152                this.blob = Some(blob);
153                // While the script waits, the rest of the simulation does not: the
154                // scheduler keeps freely choosing which *other* ticks run. The test body
155                // is re-polled after every scheduler step; the waker is only needed for
156                // the parked (quiescent) case.
157                ctx.push_park_waker(cx.waker());
158                Poll::Pending
159            }
160            Err(message) => panic!("{}", message),
161        }
162    }
163}

165/// A `pause_until` wait: resolves once the hook's pending-input status satisfies the
166/// predicate, un-pausing the hook. The status is read from the hook **on demand** at every
167/// poll (the test body is polled between every pair of scheduler steps), so the wait
168/// resolves at the first scheduling point where the predicate holds. Panics (at the
169/// `.await`'s location) if the simulation can no longer satisfy it.
170#[must_use = "the pause is only released once this future is awaited"]
171pub struct PauseUntilFuture<S, F> {
172    hook_id: usize,
173    /// The cluster member whose hook instance is paused; `None` for hooks on processes.
174    member: Option<u32>,
175    /// What the wait is called in error messages (e.g. `pause_until_count(3)`).
176    label: String,
177    predicate: F,
178    _status: PhantomData<fn(S)>,
179}
180
181impl<S: DeserializeOwned, F: Fn(&S) -> bool + Unpin> Future for PauseUntilFuture<S, F> {
182    type Output = ();
183
184    #[track_caller]
185    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
186        let this = self.get_mut();
187        let ctx = script_ctx();
188
189        // A `pause_until` wait is a script barrier, like an output await: it must not
190        // resolve (dropping the hold) while an earlier decision group is unconsumed,
191        // or the next decision would spuriously overlap the outstanding group and the
192        // boundary scan would see the exposed hook mid-group. And if that group is
193        // stuck, it is the root cause — report it instead of blaming the predicate.
194        if let Some(stuck) = script_unconsumed_description() {
195            assert!(!ctx.is_quiescent(), "{}", script_stuck_error(&stuck));
196            ctx.push_park_waker(cx.waker());
197            return Poll::Pending;
198        }
199
200        let hook = ctx.control(this.hook_id, this.member);
201
202        let status: S = bincode::deserialize(&hook.borrow().status_blob())
203            .expect("internal error: hook status blob did not match the handle's status type");
204
205        if (this.predicate)(&status) {
206            // The wait is satisfied; the hook is unpaused (the ordinary missing-decision
207            // error applies from here on).
208            hook.borrow_mut().release_hold();
209            Poll::Ready(())
210        } else if ctx.is_quiescent() {
211            let hook = hook.borrow();
212            let loc = hook.location_meta().location;
213            let member = this
214                .member
215                .map(|m| format!(" (cluster member {m})"))
216                .unwrap_or_default();
217            panic!(
218                "{} can never be satisfied: the hook at {}{} has {} and the simulation has no more work it can do",
219                this.label,
220                loc,
221                member,
222                hook.describe_pending()
223                    .unwrap_or_else(|| "no pending input".to_owned()),
224            );
225        } else {
226            ctx.push_park_waker(cx.waker());
227            Poll::Pending
228        }
229    }
230}

232/// RAII guard for [`BatchHook::pause_while`] / [`SnapshotHook::pause_while`]: ends the
233/// hold when dropped (even on panic), leaving a standing `auto_pause` hold in place.
234struct PauseGuard {
235    hook_id: usize,
236    member: Option<u32>,
237}
238
239impl Drop for PauseGuard {
240    fn drop(&mut self) {
241        let ctx = script_ctx();
242        let hook = ctx.control(self.hook_id, self.member);
243        hook.borrow_mut().release_hold();
244    }
245}

247macro_rules! pause_family {
248    ($status:ty) => {
249        /// Declares that buffering at this operator is intended: while paused, the hook is
250        /// exempt from the missing-decision error, never causes its tick to run, and — if
251        /// its tick runs anyway because *other* hooks feed it — contributes its "nothing
252        /// new" behavior each time. Scripting any decision implicitly resumes the hook.
253        ///
254        /// A pause takes its place in the script like everything else: requested while a
255        /// decision is still pending, the hold begins once that decision has been
256        /// consumed.
257        pub fn pause(&self) {
258            let ctx = script_ctx();
259            ctx.control(self.id, self.member)
260                .borrow_mut()
261                .set_hold(true);
262        }
263
264        /// Ends a [`Self::pause`] (and clears [`Self::auto_pause`] mode).
265        pub fn resume(&self) {
266            let ctx = script_ctx();
267            let hook = ctx.control(self.id, self.member);
268            let mut hook = hook.borrow_mut();
269            hook.set_auto_pause(false);
270            hook.set_hold(false);
271        }
272
273        /// Sets a standing mode where this hook only ever acts when scripted: it holds
274        /// immediately, and every scripted decision leaves a fresh hold in place behind
275        /// it.
276        ///
277        /// This deliberately opts out of the forgotten-hook protection: if the test
278        /// forgets a step, the operator silently holds its data instead of failing. The
279        /// one `auto_pause()` line at the top of a test is the reviewer-visible marker
280        /// that this hook's timing is entirely script-driven, missed steps and all.
281        pub fn auto_pause(&self) {
282            let ctx = script_ctx();
283            let hook = ctx.control(self.id, self.member);
284            let mut hook = hook.borrow_mut();
285            hook.set_auto_pause(true);
286            hook.set_hold(true);
287        }
288
289        /// Pauses the hook exactly for the duration of `body` (resuming even on panic), so
290        /// a bracketed buffering phase cannot leak a paused hook.
291        pub async fn pause_while<Fut: Future>(&self, body: Fut) -> Fut::Output {
292            self.pause();
293            let _guard = PauseGuard {
294                hook_id: self.id,
295                member: self.member,
296            };
297            body.await
298        }
299
300        /// Pauses the hook and returns a future that resolves once the hook's
301        /// pending-input status satisfies `predicate` — a synchronization point for
302        /// scripts where the right decision is not knowable upfront. The status is read
303        /// on demand at every scheduling point. After the future resolves, the hook is
304        /// unpaused; the ordinary missing-decision error applies from there on.
305        pub fn pause_until(
306            &self,
307            predicate: impl Fn(&$status) -> bool + Unpin,
308        ) -> PauseUntilFuture<$status, impl Fn(&$status) -> bool + Unpin> {
309            self.pause_until_labeled("pause_until(..)".to_owned(), predicate)
310        }
311
312        fn pause_until_labeled<F: Fn(&$status) -> bool + Unpin>(
313            &self,
314            label: String,
315            predicate: F,
316        ) -> PauseUntilFuture<$status, F> {
317            let ctx = script_ctx();
318            ctx.control(self.id, self.member)
319                .borrow_mut()
320                .set_hold(true);
321            PauseUntilFuture {
322                hook_id: self.id,
323                member: self.member,
324                label,
325                predicate,
326                _status: PhantomData,
327            }
328        }
329    };
330}

452impl<T, Scope: ScriptableHookScope> OrderingHook<T, Unbounded, AtLeastOnce, Scope>
453where
454    T: Serialize + DeserializeOwned,
455{
456    /// Scripts a top-level `assume_ordering` action on an at-least-once stream to release
457    /// one **slot** for the value `value`, *keeping* it buffered: later decisions may
458    /// emit it again (a delayed redelivery, still `AtLeastOnce`), after other elements
459    /// have taken slots in between.
460    ///
461    /// The value remains pending input, so the script must eventually retire it with
462    /// [`Self::emit_final`] (or declare the buffering with the pause family). The
463    /// immediately-following decision may name the same value again only when two or
464    /// more buffered elements share it (the adjacent slots then belong to different
465    /// elements): a sole element in adjacent slots would be a redundant duplicate —
466    /// adjacent duplicate slots denote the same at-least-once collection, and per-slot
467    /// *cardinality* belongs to the downstream `assume_retries` observation.
468    pub fn emit(&self, value: T) -> DecisionFuture {
469        DecisionFuture::new(
470            self.id,
471            self.member,
472            &AtLeastOnceOrderingDecision::Emit(value),
473        )
474    }
475
476    /// Scripts a top-level `assume_ordering` action on an at-least-once stream to release
477    /// the *last* slot for the value `value`, consuming **every** buffered element equal
478    /// to it (the script names values, not individual elements, so retirement is
479    /// value-level). Because each buffered element must occupy at least one slot,
480    /// retiring a value that several buffered elements share requires enough prior
481    /// [`Self::emit`]s: `k` equal buffered elements need at least `k` slots in total. A
482    /// value that was never emitted before simply occupies a single slot.
483    pub fn emit_final(&self, value: T) -> DecisionFuture {
484        DecisionFuture::new(
485            self.id,
486            self.member,
487            &AtLeastOnceOrderingDecision::EmitFinal(value),

541impl<T, Scope: ScriptableHookScope> RetriesHook<T, NoOrder, Unbounded, Scope>
542where
543    T: Serialize + DeserializeOwned,
544{
545    /// Scripts a top-level `assume_retries` action to release `copies` copies of the
546    /// buffered element equal to `value`, consuming it. This is where the simulator
547    /// injects the duplicates the `AtLeastOnce` input type says downstream must
548    /// tolerate; `copies` is the *total* number of deliveries, so `release(v, 1)`
549    /// observes `v` exactly once, with no duplicate.
550    ///
551    /// Any buffered element may be named — the copies' placement is unobservable on a
552    /// `NoOrder` output, so the cardinality is the entire decision.
553    ///
554    /// # Panics
555    /// Panics immediately if `copies` is zero: releasing an element zero times would be
556    /// a *loss*, which `AtLeastOnce` does not permit.
557    #[track_caller]
558    pub fn release(&self, value: T, copies: usize) -> DecisionFuture {
559        assert_copies_at_least_one(copies);
560        DecisionFuture::new(
561            self.id,
562            self.member,
563            &NoOrderRetriesDecision::Release(value, copies),
564        )
565    }
566}

568impl<T, Scope: ScriptableHookScope> RetriesHook<T, TotalOrder, Unbounded, Scope>
569where
570    T: Serialize + DeserializeOwned,
571{
572    /// Scripts a top-level `assume_retries` action to release these values, which must
573    /// match a prefix of the buffered queue in arrival order, where a value may be
574    /// repeated back-to-back to simulate retries. This is where the simulator injects
575    /// the duplicates the `AtLeastOnce` input type says downstream must tolerate. The
576    /// decision fires only once it can be honored in full; matched elements before the
577    /// last value are consumed (a later value can only start once the earlier one's
578    /// copies ended).
579    ///
580    /// This is the **non-final** form: the trailing value's element stays buffered with
581    /// its copies left open, so later decisions may release more copies of it —
582    /// scripting `release` never over-commits to an element being done. The element
583    /// remains pending input until a [`Self::release_final`] consumes it (or the
584    /// buffering is declared with the pause family). Elements left un-released by a
585    /// partial decision likewise stay pending, and downstream tick-input hooks batch
586    /// independently of how releases are grouped into decisions.
587    ///
588    /// # Panics
589    /// Panics immediately if `values` is empty: a decision must release something (use
590    /// the pause family to hold data on purpose).
591    #[track_caller]
592    pub fn release(&self, values: impl IntoIterator<Item = T>) -> DecisionFuture {
593        let values: Vec<T> = values.into_iter().collect();
594        assert!(
595            !values.is_empty(),
596            "release: at least one value must be released (declare intentional buffering with the pause family instead)"
597        );
598        DecisionFuture::new(
599            self.id,
600            self.member,
601            &OrderedRetriesDecision::Release(values),
602        )
603    }
604
605    /// Like [`Self::release`], but **final**: the trailing value's element is consumed
606    /// along with the rest of the matched prefix, declaring that no more copies of it
607    /// will be observed. With no values, finalizes a front element whose copies were
608    /// all released by earlier `release` decisions, delivering nothing more.
609    pub fn release_final(&self, values: impl IntoIterator<Item = T>) -> DecisionFuture {
610        DecisionFuture::new(
611            self.id,
612            self.member,
613            &OrderedRetriesDecision::ReleaseFinal(values.into_iter().collect()),
614        )
615    }
616
617    /// Scripts a top-level `assume_retries` action to release `copies` copies of the
618    /// front of the buffered queue, whatever it is, consuming it (waiting for an
619    /// element to arrive if the queue is empty). Unlike [`Self::release_final`], this
620    /// does not assert the released value; prefer the value-naming forms, which fail
621    /// loudly when the script falls out of sync with the program.
622    ///
623    /// # Panics
624    /// Panics immediately if `copies` is zero: releasing an element zero times would be
625    /// a *loss*, which `AtLeastOnce` does not permit.
626    #[track_caller]
627    pub fn release_next(&self, copies: usize) -> DecisionFuture {
628        assert_copies_at_least_one(copies);
629        DecisionFuture::new(
630            self.id,
631            self.member,
632            &OrderedRetriesDecision::<T>::ReleaseNext(copies),
633        )
634    }
635}
