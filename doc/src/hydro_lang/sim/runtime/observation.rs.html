1//! Top-level observation hooks ([`ObservationHook`]): hooks with no tick DFIR that are
2//! their own scheduling unit — running one *is* releasing data (e.g. `assume_ordering`
3//! on a non-tick stream, or a hooked top-level `fold`). Each is its own
4//! `SimObservation` in the scheduler. Their scripted decision/status types and
5//! [`ScriptableHook`] impls live alongside them.
6//!
7//! The retry kinds ([`TopLevelNoOrderStreamRetriesHook`], [`TopLevelOrderedStreamRetriesHook`],
8//! [`TopLevelAtLeastOnceOrderHook`]) implement only the scriptable surface, never
9//! [`ObservationHook`]: their decision spaces are infinite (every element admits
10//! arbitrarily many retries), so no autonomous exploration is possible and the builder
11//! requires them to be bound to a sim hook.
12

29/// Top-level (outside-tick) `assume_ordering` hooks release elements **one at a
30/// time** rather than shuffling the entire batch. This is the key mechanism for
31/// simulating causality in feedback cycles: when data flows through a network hop
32/// and cycles back (e.g. via `forward_ref`), the cycled-back result can arrive
33/// and interleave with elements that are still pending in the input queue.
34///
35/// For example, given input `[1, 2, 3]` where each element is mapped and sent
36/// through a network cycle, the simulator can explore orderings like
37/// `[1, map(1), 2, map(2), 3, ...]` -- the cycled-back `map(1)` arrives before `2`.
38///
39/// This is the only place where such causality is observable, because top-level
40/// unbounded streams are "maximally async" -- any non-atomic stream can be
41/// arbitrarily decoupled. The in-tick variants (`StreamOrderHook`,
42/// `KeyedStreamOrderHook`) shuffle the full batch instead, since within a tick
43/// all data is available simultaneously.
44///
45/// The `sim_top_level_assume_ordering_*` tests are regression tests ensuring
46/// this one-at-a-time release behavior correctly explores causal interleavings.
47pub struct TopLevelStreamOrderHook<T> {
48    pub input: Rc<RefCell<VecDeque<T>>>,
49    pub to_release: Option<Vec<T>>,
50    pub output: Sender<T>,
51    pub location: HookLocationMeta,
52    pub format_item_debug: fn(&T) -> Option<String>,
53}
54
55impl<T> RuntimeHook for TopLevelStreamOrderHook<T> {
56    fn has_pending_input(&self) -> bool {
57        !self.input.borrow().is_empty()
58    }
59
60    fn only_one_possible_decision(&self) -> bool {
61        // A sole buffered element has exactly one possible release; ordering only
62        // becomes a choice with two or more.
63        self.input.borrow().len() <= 1
64    }
65
66    fn release_decision(&mut self, log_writer: Option<&mut dyn std::fmt::Write>) {
67        if let Some(to_release) = self.to_release.take() {
68            if !to_release.is_empty()
69                && let Some(log_writer) = log_writer
70            {
71                let HookLocationMeta {
72                    location: batch_location,
73                    line,
74                    caret_indent,
75                } = self.location;
76                let note_str = format!(
77                    "^ observed non-deterministic order: {:?}",
78                    TruncatedVecDebug(
79                        RefCell::new(Some(to_release.iter())),
80                        8,
81                        self.format_item_debug
82                    )
83                );
84
85                let _ = writeln!(log_writer);
86                log_release(
87                    log_writer,
88                    batch_location,
89                    line,
90                    caret_indent,
91                    &note_str,
92                    colored::Color::Green,
93                );
94            }
95
96            for item in to_release {
97                self.output.try_send(item).unwrap();
98            }
99        } else {
100            panic!("No decision to release");
101        }
102    }
103
104    fn location_meta(&self) -> HookLocationMeta {
105        self.location
106    }
107}

109impl<T> ObservationHook for TopLevelStreamOrderHook<T> {
110    fn autonomous_decision<'a>(&mut self, driver: &mut Borrowed<'a>) {
111        let mut current_input = self.input.borrow_mut();
112        // Instead of a full shuffle, we only release one element at a time
113        // in order to handle possible feedback cycles.
114        let idx = (0..current_input.len()).generate(driver).unwrap();
115        let item = current_input.remove(idx).unwrap();
116        self.to_release = Some(vec![item]);
117    }
118}

252impl<T> ObservationHook for TopLevelFoldHook<T> {
253    fn autonomous_decision<'a>(&mut self, driver: &mut Borrowed<'a>) {
254        let mut current_input = self.input.borrow_mut();
255
256        // Select a non-empty subset: for each element, decide include/exclude.
257        // Only force inclusion on the last element if nothing was selected yet.
258        let mut selected = Vec::new();
259        let mut remaining = VecDeque::new();
260
261        let len = current_input.len();
262        for (i, item) in current_input.drain(..).enumerate() {
263            let is_last = i == len - 1;
264            let must_include = is_last && selected.is_empty();
265            if must_include || produce().generate(driver).unwrap() {
266                selected.push(item);
267            } else {
268                remaining.push_back(item);
269            }
270        }
271
272        // Put unselected elements back
273        *current_input = remaining;
274
275        // Always permute selected elements (Fisher-Yates) to explore all orderings.
276        // Even if commutativity is claimed via manual_proof!, the simulator is
277        // conservative and does not trust it — it still explores permutations.
278        {
279            let slen = selected.len();
280            for i in (1..slen).rev() {
281                let j = (0..=i).generate(driver).unwrap();
282                selected.swap(i, j);
283            }
284        }
285
286        self.to_release = Some(selected);
287    }
288}

410impl<K: Hash + Eq + Clone, V> ObservationHook for TopLevelKeyedStreamOrderHook<K, V> {
411    fn autonomous_decision<'a>(&mut self, driver: &mut Borrowed<'a>) {
412        let mut current_input = self.input.borrow_mut();
413
414        // Collect non-empty keys with their queue lengths
415        #[expect(clippy::disallowed_methods, reason = "FxHasher is deterministic")]
416        let nonempty_keys: Vec<(K, usize)> = current_input
417            .iter()
418            .filter(|(_, q)| !q.is_empty())
419            .map(|(k, q)| (k.clone(), q.len()))
420            .collect();
421
422        // Pick which key to release from
423        let key_idx = (0..nonempty_keys.len()).generate(driver).unwrap();
424        let (key, queue_len) = &nonempty_keys[key_idx];
425
426        // Pick which item from that key's queue
427        let item_idx = (0..*queue_len).generate(driver).unwrap();
428        let item = current_input
429            .get_mut(key)
430            .unwrap()
431            .remove(item_idx)
432            .unwrap();
433
434        self.to_release = Some(vec![(key.clone(), item)]);
435    }
436}

438/// Keyed variant of [`TopLevelStreamOrderHook`]'s scripting: a top-level keyed
439/// `assume_ordering` decision names one `(key, value)` entry to release; values within a
440/// key may be matched at any buffered position (the input is unordered within each key).
441impl<K, V> ScriptableHook for TopLevelKeyedStreamOrderHook<K, V>
442where
443    K: Hash + Eq + Clone + serde::Serialize + serde::de::DeserializeOwned,

451        Ok(self
452            .input

587/// Scripting for a top-level `entries_partially_ordered`: each decision names one
588/// `(key, value)` entry to release, and the value must be the *front* of that key's
589/// buffered queue since within-key order is preserved.
590impl<K, V> ScriptableHook for TopLevelPartiallyOrderedStreamHook<K, V>
591where
592    K: Hash + Eq + Clone + serde::Serialize + serde::de::DeserializeOwned,
593    V: serde::Serialize + serde::de::DeserializeOwned + PartialEq,
594{
595    type Decision = TopLevelOrderingDecision<(K, V)>;
596    type Status = OrderingStatus;
597
598    fn is_honorable(&self, decision: &Self::Decision) -> Result<bool, String> {
599        let TopLevelOrderingDecision::Next((key, expected)) = decision;
600        match self.input.borrow().get(key).and_then(VecDeque::front) {
601            Some(front) if front == expected => Ok(true),
602            // Within-key order is preserved, so a mismatching front can never be
603            // released ahead of the named value: the decision is permanently stuck.
604            Some(_) => Err(
605                "next: the named value did not match the front of its key's buffered queue \
606                 (within-key order is preserved by this operator)"
607                    .to_owned(),
608            ),
609            None => Ok(false),
610        }
611    }
612
613    fn apply(&mut self, decision: Self::Decision) {
614        let TopLevelOrderingDecision::Next((key, _expected)) = decision;
615        let mut input = self.input.borrow_mut();
616        let item = input.get_mut(&key).unwrap().pop_front().unwrap();
617        self.to_release = Some(vec![(key, item)]);
618    }
619
620    fn implicit(&mut self) {
621        // Implicit behavior exists for tick hooks whose tick is forced to run by *other*
622        // hooks; a top-level observation consists of exactly this hook, so it can never
623        // be forced to run without a scripted decision.
624        abort!("implicit decision invoked on a top-level partially-ordered hook");
625    }
626
627    fn status(&self) -> Self::Status {
628        OrderingStatus {
629            buffered: keyed_buffer_len(&self.input.borrow()),
630        }
631    }
632
633    fn describe_pending(&self) -> Option<String> {
634        describe_keyed_pending(&self.input.borrow())
635    }
636}

741/// released element comes from. The element may be named ([`Self::First`] /
742/// [`Self::Second`], asserting it equals the front of that input's buffer) or released

753    FirstNext(S),
754    /// Release the front of the second (right) input's buffer, whatever it is.
755    SecondNext(S),
756}

964                    candidates.push((true, k.clone()));
965                }
966            }

968

970        let (take_second, key) = &candidates[idx];
971        let take_second = *take_second;

974            self.second
975                .borrow_mut()
976                .get_mut(key)
977                .unwrap()

979                .unwrap()

982                .borrow_mut()
983                .get_mut(key)
984                .unwrap()
985                .pop_front()
986                .unwrap()
987        };

1075/// Top-level hook for `assume_retries` (`AtLeastOnce` → `ExactlyOnce`) on a **`NoOrder`**
1076/// stream: the point where the simulator injects the retries the type says downstream
1077/// must tolerate. Each `release(value, copies)` decision picks one buffered element and
1078/// a copy count `copies >= 1`, releasing that many copies and consuming the element.
1079///
1080/// The copy count is the *entire* decision: the output stays `NoOrder`, so the placement
1081/// of the copies is unobservable — downstream ordering points shuffle them like any
1082/// other elements. Releases stay one element per decision because timing matters on an
1083/// unordered stream: a released element can cycle back through feedback and be chosen
1084/// ahead of still-buffered input. This is what makes `assume_retries().assume_ordering()`
1085/// explore the same state space as `assume_ordering().assume_retries()` (see
1086/// [`TopLevelAtLeastOnceOrderHook`] for the other factoring): here each instance's
1087/// copies become distinct elements of a `NoOrder + ExactlyOnce` multiset, and the
1088/// downstream ordering observation arranges that multiset arbitrarily.

1091/// decision space infinite, so this hook exists only bound to a sim hook handle.

1100/// Top-level hook for `assume_retries` (`AtLeastOnce` → `ExactlyOnce`) on a
1101/// **`TotalOrder`** stream. A decision supplies the released sequence for a **prefix**
1102/// of the buffered queue: the prefix's values in order, where each value may be
1103/// repeated back-to-back to simulate retries.
1104///
1105/// Unlike the unordered kinds, there is no one-element-at-a-time protocol here: on an
1106/// ordered stream a feedback-cycled element always queues *behind* everything pending
1107/// (it cannot race ahead of buffered input), and downstream tick-input hooks choose
1108/// their own batch boundaries regardless of arrival granularity, so releasing several
1109/// elements in one observation forgoes no reachable behavior. A decision fires only
1110/// once it is honorable *in full* against the buffered queue.
1111///
1112/// Finality is explicit, exactly as for [`TopLevelAtLeastOnceOrderHook`]'s
1113/// `emit`/`emit_final`: elements *inside* the sequence are finalized by the sequence
1114/// itself (a later value can only start once the earlier one's copies ended — order is
1115/// preserved), but the **trailing** element's fate is the script's declaration.
1116/// `release(values)` keeps it queued (`front_started`) so later decisions may release
1117/// more copies of it, while `release_final(values)` consumes it — and, as always, any
1118/// queued element leaves the hook subject to the forgotten-hook check until a decision
1119/// or pause accounts for it. When equal elements sit next to each other in the queue,
1120/// scripted repeats are attributed to as many distinct queue elements as possible
1121/// (equal elements are indistinguishable to the script, so this is unobservable).
1122///
1123/// Back-to-back-only expansion is deliberate. An `AtLeastOnce` collection `[A, B]`
1124/// denotes the physical realizations `A⁺B⁺` (each element instance stutters *in
1125/// place*); a delayed redelivery like `ABAB` is a *different* collection
1126/// (`[A, B, A, B]`), mintable only where the *order* dimension is observed —
1127/// `assume_ordering` on an `AtLeastOnce` stream ([`TopLevelAtLeastOnceOrderHook`]) can
1128/// emit an instance into several non-adjacent slots, which then arrive here as separate
1129/// slots to expand. This factoring makes `assume_ordering().assume_retries()` and
1130/// `assume_retries().assume_ordering()` explore the same state space: any sequence in
1131/// which every input instance appears at least once.

1388/// The result of matching a scripted release sequence against an ordered retries
1389/// hook's buffered queue (see [`walk_ordered_retries`]).
1390enum PrefixWalk {
1391    /// The whole sequence matched, fully finalizing the first `consumed` queue elements
1392    /// (the trailing matched element, if any, is *not* counted: its fate is the
1393    /// decision's finality flag).
1394    Complete { consumed: usize },
1395    /// The sequence ran past the buffered queue; it may become honorable once more
1396    /// input arrives.
1397    NeedMoreInput,
1398    /// A scripted value contradicts the buffered queue; the decision can never fire
1399    /// (input order is preserved).
1400    Mismatch,
1401}
1402
1403/// Matches `values` against the queue: each value either continues the current element
1404/// (a back-to-back repeat) or starts the next queued element. `front_started` carries
1405/// over from an earlier non-final decision that left the front element's copies open.
1406/// When a value could do either (equal elements adjacent in the queue), it starts the
1407/// next element — repeats are attributed to as many distinct queue elements as
1408/// possible, which is unobservable (the elements are equal) but keeps the accounting
1409/// canonical.
1410fn walk_ordered_retries<T: PartialEq>(
1411    input: &VecDeque<T>,
1412    front_started: bool,
1413    values: &[T],
1414) -> PrefixWalk {
1415    let mut consumed = 0;
1416    let mut started = front_started;
1417    for value in values {
1418        if !started {
1419            match input.get(consumed) {
1420                Some(front) if front == value => started = true,
1421                Some(_) => return PrefixWalk::Mismatch,
1422                None => return PrefixWalk::NeedMoreInput,
1423            }
1424        } else if input.get(consumed + 1).is_some_and(|next| next == value) {
1425            // Starts the next queued element, finalizing the current one.
1426            consumed += 1;
1427        } else if input.get(consumed).is_some_and(|current| current == value) {
1428            // Another back-to-back copy of the current element.
1429        } else if consumed + 1 >= input.len() {
1430            return PrefixWalk::NeedMoreInput;
1431        } else {
1432            return PrefixWalk::Mismatch;
1433        }
1434    }
1435    PrefixWalk::Complete { consumed }
1436}

1438impl<T> ScriptableHook for TopLevelOrderedStreamRetriesHook<T>
1439where
1440    T: Clone + serde::Serialize + serde::de::DeserializeOwned + PartialEq,
1441{
1442    type Decision = OrderedRetriesDecision<T>;
1443    type Status = OrderingStatus;
1444
1445    fn is_honorable(&self, decision: &Self::Decision) -> Result<bool, String> {
1446        let values = match decision {
1447            OrderedRetriesDecision::Release(values) => {
1448                if values.is_empty() {
1449                    // The handle asserts this at the scripting call site; kept as a
1450                    // defensive permanent error.
1451                    return Err(
1452                        "a non-final release must contain at least one value (declare \
1453                         intentional buffering with the pause family instead)"
1454                            .to_owned(),
1455                    );
1456                }
1457                values
1458            }
1459            OrderedRetriesDecision::ReleaseFinal(values) => {
1460                if values.is_empty() {
1461                    return if self.front_started {
1462                        Ok(true)
1463                    } else {
1464                        Err(
1465                            "release_final with no values requires an earlier release to \
1466                             have left the front element's copies open"
1467                                .to_owned(),
1468                        )
1469                    };
1470                }
1471                values
1472            }
1473            OrderedRetriesDecision::ReleaseNext(copies) => {
1474                check_copies_at_least_one(*copies)?;
1475                return Ok(!self.input.borrow().is_empty());
1476            }
1477        };
1478        match walk_ordered_retries(&self.input.borrow(), self.front_started, values) {
1479            PrefixWalk::Complete { .. } => Ok(true),
1480            PrefixWalk::NeedMoreInput => Ok(false),
1481            // Input order is preserved, so a contradicting value can never be released:
1482            // the decision is permanently stuck.
1483            PrefixWalk::Mismatch => Err(
1484                "the scripted values did not match the buffered queue (input order is \
1485                 preserved, so values are released in arrival order and repeat only \
1486                 back-to-back)"
1487                    .to_owned(),
1488            ),
1489        }
1490    }
1491
1492    fn apply(&mut self, decision: Self::Decision) {
1493        match decision {
1494            OrderedRetriesDecision::Release(values) => {
1495                let PrefixWalk::Complete { consumed } =
1496                    walk_ordered_retries(&self.input.borrow(), self.front_started, &values)
1497                else {
1498                    abort!("ordered retries decision applied while not honorable");
1499                };
1500                self.input.borrow_mut().drain(..consumed);
1501                self.to_release = Some(values);
1502                self.release_provisional = true;
1503                self.front_started = true;
1504            }
1505            OrderedRetriesDecision::ReleaseFinal(values) => {
1506                if values.is_empty() {
1507                    // Finalizes a front element whose copies were all released by
1508                    // earlier decisions; nothing more is delivered.
1509                    self.input.borrow_mut().pop_front().unwrap();
1510                    self.to_release = Some(vec![]);
1511                } else {
1512                    let PrefixWalk::Complete { consumed } =
1513                        walk_ordered_retries(&self.input.borrow(), self.front_started, &values)
1514                    else {
1515                        abort!("ordered retries decision applied while not honorable");
1516                    };
1517                    // The trailing matched element is finalized too.
1518                    self.input.borrow_mut().drain(..=consumed);
1519                    self.to_release = Some(values);
1520                }
1521                self.release_provisional = false;
1522                self.front_started = false;
1523            }
1524            OrderedRetriesDecision::ReleaseNext(copies) => {
1525                let item = self.input.borrow_mut().pop_front().unwrap();
1526                self.to_release = Some(expand_retries(item, copies));
1527                self.release_provisional = false;
1528                self.front_started = false;
1529            }
1530        }
1531    }
1532
1533    fn implicit(&mut self) {
1534        // Implicit behavior exists for tick hooks whose tick is forced to run by *other*
1535        // hooks; a top-level observation consists of exactly this hook, so it can never
1536        // be forced to run without a scripted decision.
1537        abort!("implicit decision invoked on a top-level retries hook");
1538    }
1539
1540    fn status(&self) -> Self::Status {
1541        OrderingStatus {
1542            buffered: self.input.borrow().len(),
1543        }
1544    }
1545
1546    fn describe_pending(&self) -> Option<String> {
1547        let input = self.input.borrow();
1548        (!input.is_empty()).then(|| {
1549            format!(
1550                "{} buffered item(s) awaiting a retries decision{}: {:?}",
1551                input.len(),
1552                if self.front_started {
1553                    " (the front element's copies are still open)"
1554                } else {
1555                    ""
1556                },
1557                TruncatedVecDebug(RefCell::new(Some(input.iter())), 8, self.format_item_debug)
1558            )
1559        })
1560    }

1568/// Top-level hook for `assume_ordering` on an `AtLeastOnce` stream. Ordering a stream
1569/// that may carry retries means choosing the sequence of **slots** each element instance
1570/// occupies, and an instance may occupy several non-adjacent slots: from a queue of
1571/// `{A, B}`, `ABAB…` is a legitimate slot sequence (each extra slot is a delayed
1572/// redelivery, still `AtLeastOnce`). So each decision releases one slot:
1573///
1574/// - `emit(value)` releases a copy and *keeps* the value buffered for future slots;
1575/// - `emit_final(value)` releases the value's last slot and consumes every buffered
1576///   element equal to it, so the queue eventually drains and the forgotten-hook check
1577///   stays meaningful.
1578///
1579/// Adjacent duplicate slots are rejected as redundant when only one buffered element
1580/// matches the value: as collections under `TotalOrder + AtLeastOnce` they are
1581/// equivalent to the single slot (`AA` and `A` both denote `A⁺`), and the *cardinality*
1582/// dimension belongs to the downstream `assume_retries` observation
1583/// ([`TopLevelOrderedStreamRetriesHook`]). Decisions name **values**, not element
1584/// instances (equal instances are indistinguishable to the script), so the hook works
1585/// at the value-class level: `emit(v)` releases one slot for the class, and
1586/// `emit_final(v)` releases its last slot while consuming *every* buffered element
1587/// equal to `v` — checking, via the per-class emission counts, that the class received
1588/// at least as many slots as it has buffered elements (each element must be observed at
1589/// least once; at-least-once permits duplicates, not losses). Two adjacent slots of the
1590/// same value are legitimate exactly when two or more buffered elements share it.
1591///
1592/// There is no autonomous ([`ObservationHook`]) implementation: re-emission makes the
1593/// decision space infinite, so this hook exists only bound to a sim hook handle.
1594pub struct TopLevelAtLeastOnceOrderHook<T> {
1595    pub input: Rc<RefCell<VecDeque<T>>>,
1596    pub to_release: Option<Vec<T>>,
1597    /// Whether the staged release is a non-final slot (its value stays buffered),
1598    /// noted in the release log.
1599    pub release_provisional: bool,
1600    /// The value released by the immediately-preceding decision, if that decision was a
1601    /// provisional `emit` (elements equal to it are still buffered). The next decision
1602    /// may name the same value only if two or more buffered elements share it —
1603    /// otherwise it would place the same sole element in adjacent slots, a redundant
1604    /// duplicate. `None` after an `emit_final` (everything equal to the released value
1605    /// was consumed, so a later equal arrival is a genuinely new element).
1606    pub last_emitted: Option<T>,
1607    /// Per value-class count of provisional slots released so far, consulted by
1608    /// `emit_final` to check that every buffered element of the class got a slot.
1609    /// An entry lives from a class's first `emit` until its `emit_final` consumes it.
1610    pub emitted_counts: Vec<(T, usize)>,
1611    pub output: Sender<T>,
1612    pub location: HookLocationMeta,
1613    pub format_item_debug: fn(&T) -> Option<String>,
1614}
1615

1693impl<T> TopLevelAtLeastOnceOrderHook<T>
1694where
1695    T: PartialEq,

1697    /// The number of buffered elements equal to `value`.
1698    fn buffered_count(&self, value: &T) -> usize {
1699        self.input
1700            .borrow()
1701            .iter()
1702            .filter(|item| *item == value)
1703            .count()
1704    }
1705
1706    /// The number of provisional slots already released for `value`'s class.
1707    fn emitted_count(&self, value: &T) -> usize {
1708        self.emitted_counts
1709            .iter()
1710            .find(|(v, _)| v == value)
1711            .map_or(0, |(_, count)| *count)
1712    }
1713
1714    /// Validates a decision naming `value` against the value-class rules. Returns
1715    /// `Ok(true)` when honorable now, `Ok(false)` when it should wait for a matching
1716    /// element to arrive, and `Err` for a mis-stated script.
1717    fn check_decision(&self, value: &T, is_final: bool) -> Result<bool, String> {
1718        let buffered = self.buffered_count(value);
1719        if buffered == 0 {
1720            return Ok(false);
1721        }
1722        if self.last_emitted.as_ref() == Some(value) && buffered < 2 {
1723            return Err(
1724                "the same value cannot be released twice in a row when only one buffered \
1725                 element matches it (back-to-back duplicates are the downstream \
1726                 `assume_retries`'s decision)"
1727                    .to_owned(),
1728            );
1729        }
1730        if is_final && self.emitted_count(value) + 1 < buffered {
1731            return Err(format!(
1732                "emit_final consumes all {} buffered element(s) equal to the value, but \
1733                 only {} would have been emitted (every element must be emitted at least \
1734                 once)",
1735                buffered,
1736                self.emitted_count(value) + 1,
1737            ));
1738        }
1739        Ok(true)
1740    }
1741}
