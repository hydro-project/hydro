1//! A non-`serde` [`SimCodec`] used by the simulator's own tests.
2//!
3//! It is public because the generated simulation dylib must be able to name it.
4
5use crate::sim::codec::SimCodec;
6
7/// A test message that does not implement `serde` traits.
8#[derive(Clone, Copy, Debug, PartialEq, Eq)]
9pub struct RawMessage {
10    pub id: u64,
11    pub value: u64,
12}
13
14/// Encodes [`RawMessage`] as little-endian bytes.
15#[derive(Clone, Copy, Debug, Default)]
16pub struct RawMessageCodec;
17
18impl SimCodec<RawMessage> for RawMessageCodec {
19    fn encode(value: &RawMessage) -> Vec<u8> {
20        let mut out = Vec::with_capacity(16);
21        out.extend_from_slice(&value.id.to_le_bytes());
22        out.extend_from_slice(&value.value.to_le_bytes());
23        out
24    }
25
26    fn decode(bytes: &[u8]) -> RawMessage {
27        let (id, value) = bytes.split_at(8);
28        RawMessage {
29            id: u64::from_le_bytes(id.try_into().unwrap()),
30            value: u64::from_le_bytes(value.try_into().unwrap()),
31        }
32    }
33}
