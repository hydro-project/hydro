1//! Carrying program panics out of the simulation dylib.
2//!
3//! A compiled simulation is a `cdylib` with its own copy of the standard library, so a panic
4//! raised by program code inside it (say, an `unwrap` in a `map` closure) is a *foreign*
5//! exception to the host. Unwinding it into a host `catch_unwind` (bolero's, or a test's) aborts
6//! the whole process with "Rust cannot catch foreign exceptions", so `#[should_panic]` and the
7//! fuzzer never see it.
8//!
9//! Generated code therefore wraps every poll of every DFIR tick in [`guard_tick`], and the
10//! program's instantiation in [`guard_runtime`]. Both run in the dylib, so their `catch_unwind`
11//! is the dylib's own and catches the panic. They lend its message to the host's
12//! [`PanicHandler`], which copies it and raises it again as an ordinary host panic with a
13//! `String` payload. That panic unwinds out of the handler, through the guard's frame (no longer
14//! catching, and dropping the dylib's copy of the message), and into the host like any other.
15//!
16//! Only the message crosses the boundary, as a borrowed `&str`, never the caught
17//! `Box<dyn Any + Send>` itself. That box's vtable (`type_id`, `drop_in_place`) lives in the
18//! dylib's image, so the host must not hold it past the dylib's lifetime, and a panic payload
19//! routinely does: in a `#[should_panic]` test, unwinding drops the `CompiledSim` (and with it
20//! the `libloading::Library`) before the test harness downcasts the payload to check `expected`.
21//! Nothing is lost by converting in the dylib: `&str` and `String` are the only payload types the
22//! host could downcast anyway (`String` is the same sysroot `alloc` type on both sides), while a
23//! type from any other crate is compiled separately for the dylib and so has a different `TypeId`
24//! than the host's.
25//!
26//! The message is lent rather than given (`&str`, not `String`) so that no heap allocation
27//! changes owner across the boundary: the host and the dylib each have their own global
28//! allocator, and freeing one side's allocation with the other's is only sound while both happen
29//! to be the default `System` allocator (see hydro-project/hydro#3379). Like the
30//! `println_handler: fn(fmt::Arguments<'_>)`, the host copies what it needs while the dylib's
31//! frame is still live.
32
33use std::any::Any;
34use std::future::Future;
35use std::panic::AssertUnwindSafe;
36
37use dfir_rs::scheduled::context::{Context, TickClosure, TickClosureFn};

39/// The host function the dylib reports a program panic to, lending its message. It raises the
40/// panic on the host, so it never returns.
41pub type PanicHandler = fn(&str) -> !;
42
43/// The message of a caught panic payload, as `panic!` formats it.
44///
45/// Runs in the dylib. The payload must be consumed here rather than handed to the host (see the
46/// module docs).
47fn panic_message(payload: Box<dyn Any + Send>) -> String {
48    if let Some(s) = payload.downcast_ref::<&'static str>() {
49        (*s).to_owned()
50    } else if let Some(s) = payload.downcast_ref::<String>() {
51        s.clone()
52    } else {
53        "simulated program panicked with a non-string payload (see the `panicked at` message printed above for its location)"
54            .to_owned()
55    }
56    // `payload` is dropped here, by the code that allocated it.
57}
58
59/// Runs `f`, reporting a panic to `handler` instead of letting it unwind out of the dylib.
60fn guarded<R>(handler: PanicHandler, f: impl FnOnce() -> R) -> R {
61    match std::panic::catch_unwind(AssertUnwindSafe(f)) {
62        Ok(r) => r,
63        Err(payload) => {
64            let message = panic_message(payload);
65            handler(&message)
66            // `handler` never returns; the host's panic unwinds back through this frame, whose
67            // landing pad drops `message` with the dylib's allocator.
68        }
69    }
70}

72/// Wraps a DFIR tick closure so a panic raised while polling it is reported to `handler`.
73/// Called from generated code, through
74/// [`Dfir::map_tick_closure`](dfir_rs::scheduled::context::Dfir::map_tick_closure).
75pub fn guard_tick<Tick>(
76    mut tick: Tick,
77    handler: PanicHandler,
78) -> impl TickClosure<Metrics = Tick::Metrics> + 'static
79where
80    Tick: TickClosure + 'static,
81{
82    TickClosureFn::new(async move |ctx: &mut Context<Tick::Metrics>| -> bool {
83        let mut fut = std::pin::pin!(tick.call_tick(ctx));
84        std::future::poll_fn(move |cx| guarded(handler, || fut.as_mut().poll(cx))).await
85    })
86}
87
88/// Runs `f` (the dylib's instantiation of the program), reporting a panic to `handler`.
89pub fn guard_runtime<R>(handler: PanicHandler, f: impl FnOnce() -> R) -> R {
90    guarded(handler, f)
91}

93/// The host's [`PanicHandler`]. Copies the message into a host-owned `String` before raising.
94pub(crate) fn host_handler(message: &str) -> ! {
95    std::panic::resume_unwind(Box::new(message.to_owned()))
96}
