1//! Handle types for **simulator hooks**: scripting the decisions of unsafe operators.
2//!
3//! Every unsafe operator (like [`Stream::batch`](crate::live_collections::Stream::batch) or
4//! [`Singleton::snapshot`](crate::live_collections::Singleton::snapshot)) takes a
5//! [`NonDet`](crate::nondet::NonDet) guard. A guard can optionally carry a **hook handle**,
6//! which lets a simulation test take manual control of the non-deterministic decision made
7//! by that operator (which elements form the next batch, which version of a piece of state
8//! a snapshot reveals, ...). See `hydro_lang::sim::hooks` for the test-side scripting API.
9//!
10//! Handles are created from the [`FlowBuilder`](crate::compile::builder::FlowBuilder) via
11//! [`FlowBuilder::sim_hook`](crate::compile::builder::FlowBuilder::sim_hook) *before* the
12//! program under test is constructed, and attached to the operator they control with the
13//! `nondet!(... hook = handle)` syntax. Handles are small and `Copy`: the same value is
14//! passed into the program during construction and used later inside the test body to
15//! script decisions.
16//!
17//! # Hook scopes
18//!
19//! Every handle type carries a **scope** parameter naming the kind of root location the
20//! hooked operator runs on, mirroring
21//! [`Location::SimHookScope`](crate::location::Location::SimHookScope):
22//!
23//! - [`OnProcess<P>`] (the default): the operator has one instance, scripted directly.
24//! - [`OnCluster<C>`]: every cluster member runs its own instance of the operator;
25//!   select the one to script with [`.on(member_id)`](BatchHook::on), which yields an
26//!   [`OnMember<C>`]-scoped handle.
27//!
28//! Operators name the scope in their `NonDet` payload type, so scope mismatches fail to
29//! compile.
30//!
31//! This module contains only the handle types themselves (plain data), so components can
32//! expose hookable signatures (e.g. `nondet_batch: NonDet<Option<BatchHook<u32>>>`, passed
33//! directly to the `batch` operator it controls) without pulling
34//! in any simulator machinery; binding a hook in a flow that is *deployed* rather than
35//! simulated is harmless metadata that non-simulator backends ignore.
36

48/// A simulator hook handle (or a set of them) that can be created in one call to
49/// [`FlowBuilder::sim_hook`](crate::compile::builder::FlowBuilder::sim_hook).
50///
51/// Individual handle types implement this trait, and a struct of handles (a component's
52/// "testing interface") can implement it by creating every field. Fields are typed
53/// `Option<...>` so the struct doubles as a composite hook payload: its [`Default`]
54/// ("no hooks") is what a plain `nondet!(...)` guard carries, while `flow.sim_hook()`
55/// fills in every handle:
56///
57/// ```rust,ignore
58/// #[derive(Clone, Copy, Default)]
59/// pub struct CounterHooks {
60///     pub batch: Option<BatchHook<u32>>,
61///     pub snapshot: Option<SnapshotHook<u64>>,
62/// }
63///
64/// impl SimHook for CounterHooks {
65///     fn create(next_id: &mut dyn FnMut() -> usize) -> Self {
66///         CounterHooks {
67///             batch: SimHook::create(next_id),
68///             snapshot: SimHook::create(next_id),
69///         }
70///     }
71/// }
72/// ```
73///
74/// Such structs nest (a field can itself be a struct of handles), and since handles are
75/// `Copy` a test can pass the struct around or destructure it freely.
76///
77/// Each handle's [`SimHook`] impl carries the trait bounds that the *scripted simulation
78/// codegen* for its operator kind requires (serde round-tripping for decisions, equality
79/// for value-naming decisions, `Hash + Eq + Clone` keys for keyed buffers). Since handles
80/// can only be created through this trait, binding a hook to an operator over unsupported
81/// types fails at the `flow.sim_hook()` call — an ordinary compile error in the test crate
82/// — instead of surfacing as a rustc failure inside the generated simulation dylib.
83pub trait SimHook {
84    /// Creates every handle in this value, allocating fresh IDs via `next_id`.
85    fn create(next_id: &mut dyn FnMut() -> usize) -> Self;
86}
87
88impl<H: SimHook> SimHook for Option<H> {
89    fn create(next_id: &mut dyn FnMut() -> usize) -> Self {
90        Some(H::create(next_id))
91    }
92}

94/// Hook scope marker: the hooked operator runs on a
95/// [`Process`](crate::location::Process) with tag `P` and has one instance, which the
96/// handle scripts directly.
97///
98/// This is the default scope of every handle type. See [the module docs](self#hook-scopes).
99pub struct OnProcess<P = ()> {
100    _phantom: PhantomData<fn(P)>,
101}
102
103/// Hook scope marker: the hooked operator runs on a
104/// [`Cluster`](crate::location::Cluster) with tag `C`, so every member runs its own
105/// independent instance of the operator.
106///
107/// Select the instance to script with [`.on(member_id)`](BatchHook::on), which yields an
108/// [`OnMember`]-scoped handle. Each member's instance makes its own decisions: a member
109/// with buffered input needs its own decision or pause.
110pub struct OnCluster<C = ()> {
111    _phantom: PhantomData<fn(C)>,
112}
113
114/// Hook scope marker: one selected member's instance of an operator running on a
115/// [`Cluster`](crate::location::Cluster) with tag `C`, produced by
116/// [`.on(member_id)`](BatchHook::on).
117///
118/// Member-scoped handles script decisions like process-scoped ones, but cannot be
119/// created or bound: operators are always bound through the unscoped [`OnCluster`]
120/// handle.
121pub struct OnMember<C = ()> {
122    _phantom: PhantomData<fn(C)>,
123}
124
125/// Hook scopes a handle can be created and bound with: [`OnProcess`] and [`OnCluster`].
126/// Operator signatures select the scope through
127/// [`Location::SimHookScope`](crate::location::Location::SimHookScope).
128#[diagnostic::on_unimplemented(
129    message = "`{Self}` is not a scope that sim hook handles can be created with",
130    note = "handles are created with the `OnProcess<P>` or `OnCluster<C>` scope of the operator they will be bound to; `OnMember` handles only arise from `.on(member_id)` at scripting time"
131)]
132#[sealed::sealed]
133pub trait BindableHookScope {}
134#[sealed::sealed]
135impl<P> BindableHookScope for OnProcess<P> {}
136#[sealed::sealed]
137impl<C> BindableHookScope for OnCluster<C> {}

139/// Hook scopes that name a single instance of the hooked operator and can therefore
140/// script decisions and pauses: [`OnProcess`] and [`OnMember`]. An [`OnCluster`]-scoped
141/// handle must first select a member with [`.on(member_id)`](BatchHook::on).
142#[diagnostic::on_unimplemented(
143    message = "a `{Self}`-scoped sim hook handle cannot script decisions",
144    note = "a hook bound to an operator running on a cluster has one independent instance per member; select the member to script with `.on(member_id)`"
145)]
146#[sealed::sealed]
147pub trait ScriptableHookScope {}
148#[sealed::sealed]
149impl<P> ScriptableHookScope for OnProcess<P> {}
150#[sealed::sealed]
151impl<C> ScriptableHookScope for OnMember<C> {}

153/// Generates the `.on(member_id)` member-selection method on [`OnCluster`]-scoped
154/// handles, shared by every handle type.
155macro_rules! on_member_method {
156    ($handle:ident < $($param:ident),* >) => {
157        /// Selects one cluster member's instance of the hooked operator, returning an
158        /// [`OnMember`]-scoped handle with the full scripting API.
159        ///
160        /// Each member's instance is scripted independently: `handle.on(0).release(2)`
161        /// scripts member 0's next batch and says nothing about the other members, each
162        /// of which needs its own decision (or pause) when it holds buffered input. A
163        /// member ID outside the cluster's sizing is reported when the scripting call
164        /// is awaited.
165        pub fn on(&self, member_id: u32) -> $handle<$($param,)* OnMember<C>> {
166            $handle {
167                id: self.id,
168                member: Some(member_id),
169                _phantom: PhantomData,
170            }
171        }
172    };
173}

270/// A hook handle controlling an `assume_ordering` operator over `T` elements, with retry
271/// guarantee `R` (mirroring the type of the stream whose ordering is assumed). `S` is the
272/// handle's [scope](self#hook-scopes).
273///
274/// For an `ExactlyOnce` stream, a top-level decision selects the next buffered element to
275/// release, and an `assume_ordering` inside a tick instead takes one exhaustive ordering
276/// of that tick's complete input. For an `AtLeastOnce` stream, ordering additionally

286        expect(dead_code, reason = "only read by the `sim`-gated scripting API")
287    )]

289    pub(crate) _phantom: PhantomData<fn(T, B, R, S)>,

337/// A hook handle controlling an `assume_retries` operator over a stream of `T` elements
338/// with ordering `O` (mirroring the type of the stream whose retries are assumed). `S`
339/// is the handle's [scope](self#hook-scopes).
340///
341/// A decision for a retries hook says how many times a buffered element is released —
342/// the point where the simulator injects the duplicates the `AtLeastOnce` type says
343/// downstream must tolerate. Because every element admits arbitrarily many retries, the
344/// decision space is infinite and this non-determinism can never be explored
345/// autonomously: an `assume_retries` under simulation **must** be bound to a hook. See
346/// `hydro_lang::sim::hooks` for the decisions offered.

348    pub(crate) id: usize,
349    /// The selected cluster member (set by `.on(member_id)` on an [`OnCluster`]-scoped
350    /// handle); `None` for process-scoped handles. Only read by the simulator's
351    /// scripting machinery (`hydro_lang::sim::hooks`), which is feature-gated.
352    #[cfg_attr(
353        not(feature = "sim"),
354        expect(
355            dead_code,
356            reason = "read only by the `sim`-gated scripting machinery; handles are plain data usable without it"
357        )
358    )]
359    pub(crate) member: Option<u32>,
360    pub(crate) _phantom: PhantomData<fn(T, O, B, S)>,
361}
