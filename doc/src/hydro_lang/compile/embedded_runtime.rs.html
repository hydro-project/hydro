19/// Returns a [`QuotedWithContext`] that references the `__cluster_self_id` runtime variable.
20pub fn embedded_cluster_self_id<'a>() -> impl QuotedWithContext<'a, TaglessMemberId, ()> + Clone + 'a
21{
22    let self_id: RuntimeData<&TaglessMemberId> = RuntimeData::new("__cluster_self_id");
23    q!(self_id.clone())
24}

26/// Returns a [`QuotedWithContext`] that references a `__membership_{idx}` runtime variable.
27pub fn embedded_cluster_membership_stream<'a>(
28    idx: usize,
29) -> impl QuotedWithContext<'a, Box<dyn Stream<Item = (TaglessMemberId, MembershipEvent)> + Unpin>, ()>
30{
31    // TODO(shadaj): change `Deploy` trait to use `syn::Expr` to avoid leaking here
32    // TODO(shadaj): this will not work if the same location reads the same membership stream multiple times
33    let var_name: &'static str = Box::leak(format!("__membership_{}", idx).into_boxed_str());
34    let membership: RuntimeData<
35        Box<dyn Stream<Item = (TaglessMemberId, MembershipEvent)> + Unpin>,
36    > = RuntimeData::new(var_name);
37    q!(membership)
38}

40/// A channel crossing the boundary of an embedded dataflow: a network channel end, a stream
41/// input, a membership stream, or an output, with its guarantees as marker types (`Ordering`
42/// and `Retries`).
43///
44/// - For a stream input or membership stream, the guarantees are what the dataflow relies on,
45///   which the host's [`Promises`] must meet.
46/// - For the sending end of a network channel, they are what the dataflow sends, which the
47///   host's [`Acknowledgements`] must cover, as for an output.
48/// - For the receiving end of a network channel, they are what the dataflow relies on: the sent
49///   stream, weakened by the channel's configuration. The host's [`Promises`] must meet them, as
50///   for a stream input.
51/// - For an output, they are what the dataflow provides. The host's [`Acknowledgements`] must
52///   cover every way the output may be weaker than ordered and exactly-once.
53///
54/// Each location's generated module has a marker type implementing this for each
55/// channel crossing it, named after its field: under `incoming` / `outgoing` for
56/// network channel ends, `inputs` for stream inputs, `membership` for membership streams, and
57/// `outputs` for outputs. Each field of the generated structs is an [`EmbeddedInputChannel`] (a stream
58/// input or membership stream), a [`EmbeddedNetworkInputChannel`] (a receiving end), a [`EmbeddedNetworkOutputChannel`] (a
59/// sending end), or an [`EmbeddedOutputChannel`] (an output) over its marker. The marker is inferred from
60/// the field. The markers of network channel ends also implement [`EmbeddedNetworkChannel`].
61pub trait EmbeddedChannel {
62    /// The ordering the host must keep (for an output or the sending end of a network channel,
63    /// the ordering the dataflow provides): `TotalOrder` or `NoOrder`. For a network channel,
64    /// `TotalOrder` means each sender's messages in the order it sent them; for an input, that
65    /// the stream's order is the order to process; for a membership stream, each member's events
66    /// in the order they happened.
67    type Ordering: Ordering;
68
69    /// The retry guarantee the host must keep (for an output or the sending end of a network
70    /// channel, the one the dataflow provides): `ExactlyOnce` (nothing delivered twice) or
71    /// `AtLeastOnce`.
72    type Retries: Retries;
73}
74
75/// What a host promises about something it supplies, each guarantee promised separately with
76/// a [`manual_proof!`](crate::prelude::manual_proof) saying why it holds. An [`EmbeddedInputChannel`] or
77/// [`EmbeddedNetworkInputChannel`] is built from a stream and the `Promises` that meet its marker's
78/// guarantees.
79pub struct Promises<O, R> {
80    _promised: PhantomData<fn() -> (O, R)>,
81}
82
83impl Promises<NoOrder, AtLeastOnce> {
84    /// Promises nothing: elements may arrive in any order, and any may be delivered more than
85    /// once.
86    pub fn new() -> Self {
87        Promises {
88            _promised: PhantomData,
89        }
90    }
91}
92
93impl Default for Promises<NoOrder, AtLeastOnce> {
94    fn default() -> Self {
95        Self::new()
96    }
97}
98
99impl<O, R> Promises<O, R> {
100    fn retag<O2, R2>(self) -> Promises<O2, R2> {
101        Promises {
102            _promised: PhantomData,
103        }
104    }
105}
106
107impl<R> Promises<NoOrder, R> {
108    /// Promises the order: for a channel, each sender's messages are delivered in the order it
109    /// sent them; for an input, the stream's order is the order to process; for a membership
110    /// stream, each member's events arrive in the order they happened.
111    pub fn total_order(self, _proof: ManualProof) -> Promises<TotalOrder, R> {
112        self.retag()
113    }
114}
115
116impl<O> Promises<O, AtLeastOnce> {
117    /// Promises that nothing is delivered twice: a channel never duplicates a message (for
118    /// example by resending it after a reconnect), and every element of an input is a distinct
119    /// event.
120    pub fn exactly_once(self, _proof: ManualProof) -> Promises<O, ExactlyOnce> {
121        self.retag()
122    }
123}

125/// A channel whose guarantees promises of ordering `O` and retries `R` meet: each guarantee the
126/// channel needs is weaker than (or the same as) the one promised. It is what [`EmbeddedInputChannel::new`]
127/// requires, and lets a reusable transport state once what it provides:
128/// `fn source<C: SatisfiedBy<TotalOrder, ExactlyOnce>>()` serves every channel that needs at
129/// most that.
130pub trait SatisfiedBy<O, R>: EmbeddedChannel {}

132impl<K, O, R> SatisfiedBy<O, R> for K
133where
134    K: EmbeddedChannel,
135    K::Ordering: OrderPromisedBy<O, K>,
136    K::Retries: RetriesPromisedBy<R, K>,
137{
138}

140/// A needed order (`Self`) that the promised order `Promised` meets, for the channel end `End`.
141/// Only order is promised, so a promise is needed only for `TotalOrder`.
142#[diagnostic::on_unimplemented(
143    message = "`{End}` relies on its elements arriving in order, but the host does not promise it",
144    label = "does not promise `TotalOrder`",
145    note = "add `.total_order(manual_proof!(/** why the order is kept */))` to its `Promises`"
146)]
147trait OrderPromisedBy<Promised, End> {}
148
149impl<Promised: Ordering, End> OrderPromisedBy<Promised, End> for NoOrder {}
150impl<End> OrderPromisedBy<TotalOrder, End> for TotalOrder {}

153#[diagnostic::on_unimplemented(
154    message = "`{End}` relies on each element arriving exactly once, but the host does not promise it",
155    label = "does not promise `ExactlyOnce`",
156    note = "add `.exactly_once(manual_proof!(/** why nothing is delivered twice */))` to its `Promises`"
157)]
158trait RetriesPromisedBy<Promised, End> {}
159

163/// A channel flowing into an embedded dataflow, as the host provides it: a `Stream` for a stream
164/// input or a membership stream, for the channel `C` (its generated marker type, see
165/// [`EmbeddedChannel`]). The receiving end of a network channel is a [`EmbeddedNetworkInputChannel`] instead.
166///
167/// An `EmbeddedInputChannel` exists only with promises that meet `C`'s guarantees:
168/// [`EmbeddedInputChannel::new`] takes the [`Promises`] and checks them, so wiring a channel always
169/// states its guarantees and a mismatch with the dataflow is a type error at that call. Promising
170/// more than a channel needs is fine. `C` is inferred from the field the `EmbeddedInputChannel` fills.
171pub struct EmbeddedInputChannel<C, S> {
172    inner: S,
173    _channel: PhantomData<fn() -> C>,
174}
175

178    pub fn new<O, R>(inner: S, promises: Promises<O, R>) -> Self
179    where
180        C: SatisfiedBy<O, R>,
181    {
182        let _ = promises;
183        EmbeddedInputChannel {
184            inner,
185            _channel: PhantomData,
186        }
187    }
188}

191    /// The host's stream. Used by generated code.

198/// The configuration of a network channel crossing the boundary of an embedded dataflow: the
199/// transport type `Transport` of the network config the dataflow sends with (for example
200/// `Tcp<FailStop>` for `TCP.fail_stop()`, or `EmbeddedTransport<ExactlyOnce, NoOrder, FailStop>`
201/// for `EMBEDDED.may_reorder().never_duplicates().never_drops()`), and its serialization `Serialization` (`Bincode`, or `Embedded`
202/// for an unserialized `EMBEDDED` channel).
203///
204/// The markers of both ends of a network channel implement it, alongside [`EmbeddedChannel`]
205/// for the stream at that end. The host acknowledges the configuration exactly at each end with
206/// a [`ChannelConfiguration`], so that changing the channel's network config fails the host's
207/// build at both ends until it acknowledges the change.
208pub trait EmbeddedNetworkChannel: EmbeddedChannel {
209    /// The transport type of the channel's network config.
210    type Transport;
211
212    /// The serialization of the channel's network config.
213    type Serialization;
214}
215
216/// The configuration of a network channel as the host acknowledges it: the transport `Tr` and
217/// serialization `S` of its network config, which must match the channel's exactly (see
218/// [`EmbeddedNetworkChannel`]). It is built from the same network config value the dataflow
219/// sends with; the config's name is ignored.
220pub struct ChannelConfiguration<Tr, S> {
221    _acknowledged: PhantomData<fn() -> (Tr, S)>,
222}
223
224impl<Tr, S> ChannelConfiguration<Tr, S> {
225    /// Acknowledges a channel configured with `config`, the same network config the dataflow
226    /// sends with (for example `TCP.fail_stop().bincode()`).
227    pub fn new<Name>(config: NetworkingConfig<Tr, S, Name>) -> Self {
228        let _ = config;
229        ChannelConfiguration {
230            _acknowledged: PhantomData,
231        }
232    }
233}

352/// What a host acknowledges about a stream the dataflow provides (an output, or the stream sent
353/// on a network channel): each way it may be weaker than ordered and exactly-once. The stream's
354/// type already says what it may do, so an acknowledgement needs no justification: calling it is
355/// the host's agreement, and a stream that becomes weaker fails the host's build until the host
356/// acknowledges the change. An
357/// [`EmbeddedOutputChannel`] or [`EmbeddedNetworkOutputChannel`] is built from a sink and the `Acknowledgements` that
358/// cover what its marker's stream may do.
359pub struct Acknowledgements<O, R> {
360    _acknowledged: PhantomData<fn() -> (O, R)>,
361}
362
363impl Acknowledgements<TotalOrder, ExactlyOnce> {
364    /// Acknowledges nothing: the sink relies on receiving elements in the dataflow's order, each
365    /// exactly once.
366    pub fn new() -> Self {
367        Acknowledgements {
368            _acknowledged: PhantomData,
369        }
370    }
371}
372
373impl Default for Acknowledgements<TotalOrder, ExactlyOnce> {
374    fn default() -> Self {
375        Self::new()
376    }
377}

380    fn retag<O2, R2>(self) -> Acknowledgements<O2, R2> {
381        Acknowledgements {
382            _acknowledged: PhantomData,
383        }
384    }
385}

389    pub fn no_order(self) -> Acknowledgements<NoOrder, R> {

391    }
392}
