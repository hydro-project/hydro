1//! An implementation of the RAFT consensus protocol for replicating a log across a
2//! cluster.
3//!
4//! The protocol logic is a pure, sequential step function over one unified per-member
5//! state (see [`raft_step`]), hosted in a single dataflow tick — deliberately *not*
6//! decomposed into separately-ticking election/replication components, because RAFT's
7//! safety proofs interlock vote and log decisions through shared state (see the design
8//! notes on [`raft`]).
9//!
10//! # Protocol mechanics
11//!
12//! Each member is a follower, a candidate, or the leader. A member whose election
13//! timer fires becomes a candidate and requests votes, sending the index and term of
14//! its last log entry; voters refuse candidates whose log is behind their own (RAFT
15//! §5.4.1), which is what guarantees an elected leader holds every committed entry.
16//!
17//! The leader replicates by sending each follower `AppendEntries` messages carrying
18//! new entries (empty for a pure heartbeat), the index and term of the entry
19//! immediately preceding them, and the leader's commit index. The follower accepts
20//! the entries iff its log contains a matching entry at that preceding position (the
21//! log-matching property, RAFT §5.3), and advances its own commit index to
22//! `min(leader_commit, index of last new entry)`. Acknowledgements report the highest
23//! index known replicated on the follower; the leader advances its commit index once
24//! a majority of members hold an entry of the current term, and steps down when any
25//! reply carries a higher term than its own.

146/// The single wire format for all intra-cluster RAFT traffic: vote RPCs and
147/// replication RPCs travel over one channel, so every message is processed by the
148/// same per-tick step against the same unified state — this is what makes the
149/// vote/ack interlock atomic (see [`raft`]).
150///
151/// Trait impls avoid derive bounds on `ClusterTag` (it only appears inside
152/// [`MemberId`], which implements everything for any tag); serde bounds constrain
153/// only `T`.
154///
155/// Public only because staged (`q!`) code must reference it by path; not stable API.
156#[doc(hidden)]
157#[derive(Serialize, Deserialize)]
158#[serde(bound(
159    serialize = "T: Serialize",
160    deserialize = "T: serde::de::DeserializeOwned"
161))]
162pub enum RaftRpc<T, ClusterTag> {
163    /// Candidate -> everyone: request a vote for `term` (RAFT §5.2).
164    RequestVote(RequestVoteDto),
165    /// Voter -> candidate: grant a vote for `term`.
166    RequestVoteResponse(RequestVoteResponseDto),
167    /// Leader -> follower: replicate entries / heartbeat (RAFT §5.3).
168    AppendEntries(AppendEntriesRequest<T, ClusterTag>),
169    /// Follower -> leader: acknowledge or reject an `AppendEntries`.
170    AppendEntriesReply(AppendEntriesReply),
171}

285/// The complete per-member RAFT state — election *and* replication — persisted
286/// across ticks by the unified server component inside [`raft`].
287///
288/// Keeping `term` / `voted_for` / `log` in one struct, mutated by one sequential
289/// step function ([`raft_step`]), is what makes RAFT's safety argument hold: the
290/// decision to grant a vote and the decision to acknowledge log entries are
291/// read-modify-writes against the *same* state, so no schedule can interleave them
292/// against stale views of each other. (A previous design split this state across
293/// two dataflow components with asynchronous feedback, which was a real,
294/// simulator-reproducible safety bug: committed entries could be truncated.)
295///
296/// Public only because staged (`q!`) code must reference it by path; not stable API.
297#[doc(hidden)]
298pub struct RaftServerState<T, ClusterTag> {
299    /// The latest term this member has observed (RAFT's `currentTerm`).
300    pub term: usize,
301    /// Who this member voted for in `term`, if anyone (itself, when a candidate).
302    pub voted_for: Option<MemberId<ClusterTag>>,
303    /// The member's current role.
304    pub role: RaftState,
305    /// The voters (including itself) that granted this member's candidacy for `term`.
306    pub votes: HashSet<MemberId<ClusterTag>>,
307    /// Whether an `AppendEntries` valid for the current term has been observed since
308    /// the previous election timer interrupt (suppresses the next election).
309    pub heartbeat_seen: bool,
310    /// The leader of `term` as learned from received `AppendEntries`, if any. Not
311    /// used while this member is itself the leader.
312    pub known_leader: Option<MemberId<ClusterTag>>,
313    /// The replicated log (1-based indexing via `LogEntry::index`).
314    pub log: Vec<LogEntry<T>>,
315    /// The highest log index known to be committed (0 = nothing committed).
316    pub commit_index: usize,
317    /// The highest log index already emitted on the `committed` output; used to emit
318    /// each committed entry exactly once, in log order.
319    pub emitted_index: usize,
320    /// Leader bookkeeping: the next log index to send to each follower. Reset on
321    /// every leadership acquisition.
322    pub next_index: HashMap<MemberId<ClusterTag>, usize>,
323    /// Leader bookkeeping: the highest log index known replicated on each follower,
324    /// for the current leadership only (acks are term-filtered). Reset on every
325    /// leadership acquisition.
326    pub match_index: HashMap<MemberId<ClusterTag>, usize>,
327}

422/// Advances one member's [`RaftServerState`] by one tick: processes the received
423/// messages, client requests, and timer events, returning the messages to send and
424/// the entries that became committed.
425///
426/// This is deliberately a *pure, sequential* function (plain Rust, no staging): every
427/// message is handled one at a time against the live state, so the protocol's
428/// interlock invariants hold by construction —
429///
430/// * a member that acknowledged entries in term `T` compares any later vote request
431///   against the log that *includes* those entries (§5.4.1 reads the real log);
432/// * a member that voted in term `T'` has `term = T'` when the next `AppendEntries`
433///   arrives, so it rejects any leader of an older term (§5.1's term check reads the
434///   real term).
435///
436/// Message processing is order-insensitive at the batch level: the batch is sorted
437/// into a canonical order first, so the outcome is a function of the batch multiset
438/// (required for deterministic simulation).
439///
440/// Public (and callable outside staged code) so protocol-rule unit tests can drive
441/// members directly and deterministically.
442#[doc(hidden)]
443pub fn raft_step<T: Clone, ClusterTag>(
444    state: &mut RaftServerState<T, ClusterTag>,
445    input: RaftStepInput<T, ClusterTag>,
446) -> RaftStepOutput<T, ClusterTag> {
447    let RaftStepInput {
448        me,
449        other_members,
450        cluster_size,
451        election_timer_fired,
452        heartbeat_timer_fired,
453        requests,
454        mut messages,
455    } = input;
456    let majority = cluster_size / 2 + 1;
457
458    let mut outbound = Vec::new();
459    let mut committed = Vec::new();
460    let mut redirected = Vec::new();
461
462    let old_view = LeaderView {
463        term: state.term,
464        leader: if state.role == RaftState::Leader {
465            Some(me.clone())
466        } else {
467            state.known_leader.clone()
468        },
469    };
470
471    // A term above ours forces us back to follower with all per-term state reset
472    // (RAFT §5.1). Returns whether the message's term is current afterwards.
473    fn observe_term<T, ClusterTag>(
474        state: &mut RaftServerState<T, ClusterTag>,
475        term: usize,
476    ) -> bool {
477        if term > state.term {
478            state.term = term;
479            state.role = RaftState::Follower;
480            state.voted_for = None;
481            state.votes.clear();
482            state.known_leader = None;
483        }
484        term == state.term
485    }
486
487    fn become_leader<T, ClusterTag>(
488        state: &mut RaftServerState<T, ClusterTag>,
489        other_members: &[MemberId<ClusterTag>],
490    ) {
491        state.role = RaftState::Leader;
492        state.known_leader = None;
493        state.next_index.clear();
494        state.match_index.clear();
495        for follower in other_members {
496            state
497                .next_index
498                .insert(follower.clone(), state.log.len() + 1);
499            state.match_index.insert(follower.clone(), 0);
500        }
501    }
502
503    // (a) Process this tick's messages in a canonical order (so the step is a
504    // function of the batch multiset), each one sequentially against live state.
505    fn sort_key<T, ClusterTag>(rpc: &RaftRpc<T, ClusterTag>) -> (u8, usize, usize, usize) {
506        match rpc {
507            RaftRpc::RequestVote(dto) => (0, dto.term, dto.last_log_term, dto.last_log_index),
508            RaftRpc::RequestVoteResponse(dto) => (1, dto.term, 0, 0),
509            RaftRpc::AppendEntries(request) => (
510                2,
511                request.term,
512                request.prev_log_index,
513                request.entries.len(),
514            ),
515            RaftRpc::AppendEntriesReply(reply) => {
516                (3, reply.term, reply.match_index, usize::from(reply.success))
517            }
518        }
519    }
520    messages.sort_by(|(sender_a, rpc_a), (sender_b, rpc_b)| {
521        sort_key(rpc_a)
522            .cmp(&sort_key(rpc_b))
523            .then_with(|| sender_a.cmp(sender_b))
524    });
525
526    for (sender, message) in messages {
527        match message {
528            RaftRpc::RequestVote(dto) => {
529                if !observe_term(state, dto.term) {
530                    // A stale-term candidacy: ignore (the candidate will observe the
531                    // higher term from other traffic or retry at its next timeout).
532                    continue;
533                }
534                // §5.4.1 election restriction: only grant if the candidate's log is
535                // at least as up-to-date as ours — this reads the *live* log, which
536                // is exactly the interlock the split-component design lacked.
537                let candidate_up_to_date =
538                    (dto.last_log_term, dto.last_log_index) >= state.last_log_position();
539                let can_vote =
540                    state.voted_for.is_none() || state.voted_for.as_ref() == Some(&sender);
541                if candidate_up_to_date && can_vote {
542                    state.voted_for = Some(sender.clone());
543                    outbound.push((
544                        sender,
545                        RaftRpc::RequestVoteResponse(RequestVoteResponseDto { term: state.term }),
546                    ));
547                }
548            }
549            RaftRpc::RequestVoteResponse(dto) => {
550                if !observe_term(state, dto.term) {
551                    continue;
552                }
553                if state.role == RaftState::Candidate {
554                    state.votes.insert(sender);
555                    if state.votes.len() >= majority {
556                        become_leader(state, &other_members);
557                    }
558                }
559            }
560            RaftRpc::AppendEntries(request) => {
561                if !observe_term(state, request.term) {
562                    // Stale leader: reject with our (higher) term so it steps down.
563                    outbound.push((
564                        sender,
565                        RaftRpc::AppendEntriesReply(AppendEntriesReply {
566                            term: state.term,
567                            success: false,
568                            match_index: 0,
569                        }),
570                    ));
571                    continue;
572                }
573                // A current-term AppendEntries while we lead the same term would
574                // mean two leaders share a term — election safety is broken.
575                assert!(
576                    state.role != RaftState::Leader,
577                    "protocol violation: two leaders share term {}",
578                    state.term,
579                );
580                // A live leader exists for the current term: suppress the next
581                // election, depose our candidacy, learn the leader's identity.
582                state.heartbeat_seen = true;
583                if state.role == RaftState::Candidate {
584                    state.role = RaftState::Follower;
585                }
586                state.known_leader = Some(request.leader.clone());
587
588                // Log-matching check (RAFT §5.3).
589                let log_matches = request.prev_log_index == 0
590                    || (state.log.len() >= request.prev_log_index
591                        && state.log[request.prev_log_index - 1].term_received
592                            == request.prev_log_term);
593                if !log_matches {
594                    outbound.push((
595                        sender,
596                        RaftRpc::AppendEntriesReply(AppendEntriesReply {
597                            term: state.term,
598                            success: false,
599                            match_index: 0,
600                        }),
601                    ));
602                    continue;
603                }
604
605                // Append: truncate on conflict, skip already-present entries.
606                let new_match = request.prev_log_index + request.entries.len();
607                for entry in request.entries {
608                    if state.log.len() >= entry.index {
609                        if state.log[entry.index - 1].term_received != entry.term_received {
610                            // A committed prefix can never legitimately conflict: the
611                            // leader completeness property (enforced by the §5.4.1
612                            // vote check above) guarantees every leader's log
613                            // contains all committed entries. Fail fast if this
614                            // invariant is ever violated.
615                            assert!(
616                                entry.index > state.commit_index,
617                                "protocol violation: asked to truncate committed \
618                                 entry at index {} (commit_index {})",
619                                entry.index,
620                                state.commit_index,
621                            );
622                            state.log.truncate(entry.index - 1);
623                            state.log.push(entry);
624                        }
625                    } else {
626                        state.log.push(entry);
627                    }
628                }
629
630                // Adopt the leader's commit index, capped at what this request
631                // confirmed matches the leader's log.
632                let commit_cap = request.leader_commit.min(new_match);
633                if commit_cap > state.commit_index {
634                    state.commit_index = commit_cap;
635                }
636
637                outbound.push((
638                    sender,
639                    RaftRpc::AppendEntriesReply(AppendEntriesReply {
640                        term: state.term,
641                        success: true,
642                        match_index: new_match,
643                    }),
644                ));
645            }
646            RaftRpc::AppendEntriesReply(reply) => {
647                if !observe_term(state, reply.term) {
648                    // An ack for an earlier leadership: the follower's log may have
649                    // been truncated since, so it must not count toward this term.
650                    continue;
651                }
652                if state.role != RaftState::Leader {
653                    continue;
654                }
655                if reply.success {
656                    // Successful acknowledgements advance match/next monotonically.
657                    let best = state.match_index.entry(sender.clone()).or_insert(0);
658                    if reply.match_index > *best {
659                        *best = reply.match_index;
660                    }
661                    let next = state.next_index.entry(sender).or_insert(1);
662                    if reply.match_index + 1 > *next {
663                        *next = reply.match_index + 1;
664                    }
665                } else {
666                    // Rejections back off next_index to retry earlier prefixes.
667                    if let Some(next) = state.next_index.get_mut(&sender) {
668                        *next = (*next - 1).max(1);
669                    }
670                }
671            }
672        }
673    }
674
675    // (b) Client requests: the leader appends them in the caller-fixed order;
676    // everyone else redirects them with the best-known leader hint.
677    for message in requests {
678        if state.role == RaftState::Leader {
679            let index = state.log.len() + 1;
680            state.log.push(LogEntry {
681                message,
682                term_received: state.term,
683                index,
684            });
685        } else {
686            redirected.push((message, state.known_leader.clone()));
687        }
688    }
689
690    // (c) Election timer interrupt. A leader runs no election timer. Otherwise: an
691    // AppendEntries observed since the previous interrupt suppresses (and consumes
692    // the window); if none, become a candidate for the next term (RAFT §5.2).
693    if election_timer_fired && state.role != RaftState::Leader {
694        if state.heartbeat_seen {
695            state.heartbeat_seen = false;
696        } else {
697            state.term += 1;
698            state.role = RaftState::Candidate;
699            state.voted_for = Some(me.clone());
700            state.votes.clear();
701            state.votes.insert(me.clone());
702            state.known_leader = None;
703            if state.votes.len() >= majority {
704                // Single-member cluster: our own vote suffices.
705                become_leader(state, &other_members);
706            } else {
707                let (last_log_term, last_log_index) = state.last_log_position();
708                for target in &other_members {
709                    outbound.push((
710                        target.clone(),
711                        RaftRpc::RequestVote(RequestVoteDto {
712                            term: state.term,
713                            last_log_index,
714                            last_log_term,
715                        }),
716                    ));
717                }
718            }
719        }
720    }
721
722    // (d) Advance the leader's commit index: the §5.4.2 rule — only entries of the
723    // leader's own term commit by counting replicas; earlier-term entries commit
724    // transitively beneath them.
725    if state.role == RaftState::Leader {
726        let mut candidate = state.log.len();
727        while candidate > state.commit_index {
728            if state.log[candidate - 1].term_received == state.term {
729                // Count acks by iterating `other_members` (a fixed, deterministic
730                // list) rather than the `match_index` map, whose iteration order is
731                // nondeterministic.
732                let acks = 1 + other_members
733                    .iter()
734                    .filter(|member| {
735                        state
736                            .match_index
737                            .get(member)
738                            .is_some_and(|match_index| *match_index >= candidate)
739                    })
740                    .count();
741                if acks >= majority {
742                    state.commit_index = candidate;
743                    break;
744                }
745            }
746            candidate -= 1;
747        }
748    }
749
750    // (e) Heartbeat timer interrupt: the leader broadcasts AppendEntries carrying
751    // each follower's outstanding suffix and the freshly advanced commit index.
752    if heartbeat_timer_fired && state.role == RaftState::Leader {
753        for follower in &other_members {
754            let next = state
755                .next_index
756                .get(follower)
757                .copied()
758                .unwrap_or(state.log.len() + 1);
759            let prev_log_index = next - 1;
760            let prev_log_term = if prev_log_index == 0 {
761                0
762            } else {
763                state.log[prev_log_index - 1].term_received
764            };
765            outbound.push((
766                follower.clone(),
767                RaftRpc::AppendEntries(AppendEntriesRequest {
768                    term: state.term,
769                    leader: me.clone(),
770                    prev_log_index,
771                    prev_log_term,
772                    entries: state.log[prev_log_index..].to_vec(),
773                    leader_commit: state.commit_index,
774                }),
775            ));
776        }
777    }
778
779    // (f) Emit newly committed entries exactly once, in log order.
780    while state.emitted_index < state.commit_index {
781        state.emitted_index += 1;
782        committed.push(state.log[state.emitted_index - 1].clone());
783    }
784
785    // (g) Report a view transition iff the view changed this tick.
786    let new_view = LeaderView {
787        term: state.term,
788        leader: if state.role == RaftState::Leader {
789            Some(me)
790        } else {
791            state.known_leader.clone()
792        },
793    };
794    let view_transition = (new_view != old_view).then_some(new_view);
795
796    RaftStepOutput {
797        outbound,
798        committed,
799        redirected,
800        view_transition,
801    }
802}

830/// The unified RAFT server: one state machine per member, advanced by [`raft_step`]
831/// inside a single tick that consumes every input the protocol reacts to (client
832/// requests, both timers, and all intra-cluster traffic over one channel).
833///
834/// See [`raft`] for the input/timer contracts. The only feedback loop — a member's
835/// outbound messages reaching other members' inputs — passes through the network
836/// demux, which is an asynchronous boundary, so no in-tick cycle exists and no
837/// tick-deferral plumbing is needed.
838pub fn raft_server<'a, T, ClusterTag, O, Net>(
839    cluster: &Cluster<'a, ClusterTag>,
840    requests: Stream<T, Cluster<'a, ClusterTag>, Unbounded, O>,
841    election_timer_interrupts: Stream<(), Cluster<'a, ClusterTag>>,
842    heartbeat_timer_interrupts: Stream<(), Cluster<'a, ClusterTag>>,
843    config: RaftConfig,
844    net: Net,
845    nondet_raft: NonDet,
846) -> RaftOutputs<'a, T, ClusterTag>
847where
848    T: Clone + Serialize + DeserializeOwned + 'a,
849    ClusterTag: 'a,
850    O: Ordering,
851    Net: NetworkFor<RaftRpc<T, ClusterTag>>,
852    NoOrder: MinOrder<Net::OrderingGuarantee, Min = NoOrder>,
853{
854    let cluster_size = config.cluster_size;
855
856    // Fix an arbitrary total order for incoming requests up front (a no-op when the
857    // input is already `TotalOrder`): the leader must place concurrent requests at
858    // *some* log indexes, and which order it picks is exactly the non-determinism
859    // acknowledged by `nondet_raft`.
860    let requests = requests.assume_ordering::<TotalOrder>(nondet!(
861        /// The arrival order of concurrent client requests at the leader determines
862        /// their log order, which is inherently non-deterministic.
863        nondet_raft
864    ));
865
866    // All intra-cluster traffic flows over a single channel. The forward_ref closes
867    // the loop: the sliced! block below both consumes received messages and produces
868    // the outbound messages that feed the channel (through the network — an async
869    // boundary, so this is not an in-tick cycle).
870    #[expect(clippy::type_complexity, reason = "forward_ref requires the full type")]
871    let (traffic_handle, traffic): (
872        ForwardHandle<
873            'a,
874            Stream<
875                (MemberId<ClusterTag>, RaftRpc<T, ClusterTag>),
876                Cluster<'a, ClusterTag>,
877                Unbounded,
878                NoOrder,
879            >,
880        >,
881        Stream<
882            (MemberId<ClusterTag>, RaftRpc<T, ClusterTag>),
883            Cluster<'a, ClusterTag>,
884            Unbounded,
885            NoOrder,
886        >,
887    ) = cluster.forward_ref();
888
889    // The cluster membership list, resolved on each member at runtime (identical
890    // everywhere, since it comes from deploy/sim metadata).
891    let LocationId::Cluster(cluster_key) = Location::id(cluster) else {
892        unreachable!("raft_server always runs on a cluster")
893    };
894    let cluster_members = ClusterIds {
895        key: cluster_key,
896        _phantom: PhantomData,
897    };
898
899    #[expect(
900        clippy::type_complexity,
901        reason = "the sliced! outputs are annotated with their full stream types"
902    )]
903    let (outbound_messages, committed, redirected, view_transitions): (
904        Stream<(MemberId<ClusterTag>, RaftRpc<T, ClusterTag>), Cluster<'a, ClusterTag>>,
905        Stream<LogEntry<T>, Cluster<'a, ClusterTag>>,
906        Stream<(T, Option<MemberId<ClusterTag>>), Cluster<'a, ClusterTag>>,
907        Stream<LeaderView<ClusterTag>, Cluster<'a, ClusterTag>>,
908    ) = sliced! {
909        let request_batch = use::batch(requests, nondet!(
910            /// Which requests are batched together only affects which log indexes the
911            /// leader assigns them, folded into the arbitrary order fixed above.
912            nondet_raft
913        ));
914        let election_batch = use::batch(election_timer_interrupts, nondet!(
915            /// When timer interrupts are processed relative to messages only affects
916            /// which elections are attempted, which folds into the non-determinism of
917            /// which member wins an election.
918            nondet_raft
919        ));
920        let heartbeat_batch = use::batch(heartbeat_timer_interrupts, nondet!(
921            /// Heartbeat timing only affects when replication progress is made, not
922            /// the committed sequence.
923            nondet_raft
924        ));
925        let traffic_batch = use::batch(traffic, nondet!(
926            /// Message delivery interleavings shift which member wins elections and
927            /// when entries replicate and commit, but never the committed sequence
928            /// itself: every message is processed atomically against the member's
929            /// full state.
930            nondet_raft
931        ));
932
933        let mut server_state = use::state(|l| l.singleton(q!(RaftServerState::new())));
934
935        let tick = request_batch.location().clone();
936        let election_fired = election_batch.count().map(q!(|n| n > 0));
937        let heartbeat_fired = heartbeat_batch.count().map(q!(|n| n > 0));
938
939        // Requests in the order fixed by assume_ordering (TotalOrder fold).
940        let request_vec = request_batch.fold(
941            q!(|| Vec::new()),
942            q!(|requests, request| {
943                requests.push(request);
944            }),
945        );
946
947        // Received messages as an unordered batch; raft_step sorts them into a
948        // canonical order, so downstream results depend only on the batch multiset.
949        let message_vec = traffic_batch.fold(
950            q!(|| Vec::new()),
951            q!(
952                |messages, message| {
953                    messages.push(message);
954                },
955                commutative = manual_proof!(
956                    /** the accumulated batch is sorted into a canonical order by
957                    raft_step before being applied, so downstream results depend only
958                    on the batch multiset, never on arrival order */
959                )
960            ),
961        );
962
963        // The other members of the cluster, the broadcast targets.
964        let other_members = tick.singleton(q!(
965            cluster_members
966                .iter()
967                .map(|id| MemberId::from_tagless(id.clone()))
968                .filter(|member| *member != CLUSTER_SELF_ID)
969                .collect::<Vec<_>>()
970        ));
971
972        // Reference handles for this tick's aggregates, the persistent state, and the
973        // side-channel outputs: reads via `by_ref` (`&T`), writes via `by_mut`
974        // (`&mut T` on the state, `&mut Vec<T>` on streams). The step is hosted on a
975        // one-element-per-tick trigger stream whose output is the outbound message
976        // stream itself — which also guarantees the step is never dead-code pruned.
977        let state_ref = server_state.by_mut();
978        let election_fired_ref = election_fired.by_ref();
979        let heartbeat_fired_ref = heartbeat_fired.by_ref();
980        let request_vec_ref = request_vec.by_ref();
981        let message_vec_ref = message_vec.by_ref();
982        let other_members_ref = other_members.by_ref();
983
984        let committed: Stream<LogEntry<T>, _, Bounded> = tick.source_iter(q!(Vec::new()));
985        let committed_ref = committed.by_mut();
986        let redirected: Stream<(T, Option<MemberId<ClusterTag>>), _, Bounded> =
987            tick.source_iter(q!(Vec::new()));
988        let redirected_ref = redirected.by_mut();
989        let view_transitions: Stream<LeaderView<ClusterTag>, _, Bounded> =
990            tick.source_iter(q!(Vec::new()));
991        let view_transitions_ref = view_transitions.by_mut();
992
993        // The entire protocol step: hand this tick's aggregates to the pure
994        // sequential state machine and distribute its outputs.
995        let outbound = tick
996            .singleton(q!(()))
997            .into_stream()
998            .flat_map_ordered(q!(move |_| {
999                let output = crate::cluster::raft::raft_step(
1000                    &mut *state_ref,
1001                    RaftStepInput {
1002                        me: CLUSTER_SELF_ID.clone(),
1003                        other_members: other_members_ref.clone(),
1004                        // Written without field shorthand: stageleft renames captured
1005                        // free variables, and shorthand hides the use site from it.
1006                        cluster_size: { cluster_size },
1007                        election_timer_fired: *election_fired_ref,
1008                        heartbeat_timer_fired: *heartbeat_fired_ref,
1009                        requests: request_vec_ref.clone(),
1010                        messages: message_vec_ref.clone(),
1011                    },
1012                );
1013                for entry in output.committed {
1014                    committed_ref.push(entry);
1015                }
1016                for redirect in output.redirected {
1017                    redirected_ref.push(redirect);
1018                }
1019                if let Some(view) = output.view_transition {
1020                    view_transitions_ref.push(view);
1021                }
1022                output.outbound
1023            }));
1024
1025        (outbound, committed, redirected, view_transitions)
1026    };
1027
1028    traffic_handle.complete(outbound_messages
1029            .into_keyed()
1030            // The channel's fault model is chosen by the caller: sim tests use
1031            // `TCP.fail_stop()` (the simulator cannot explore lossy channels without
1032            // giving up liveness assertions), while real deployments that face
1033            // partitions pass `TCP.lossy_delayed_forever()` — RAFT tolerates the
1034            // loss: lost AppendEntries (or replies) are re-sent on the next
1035            // heartbeat, and lost vote traffic is retried at the next election
1036            // timeout.
1037            .demux(cluster, net)
1038            .entries());
1039
1040    RaftOutputs {
1041        // The RAFT protocol itself is what justifies the consistency cast: every
1042        // member emits the same totally-ordered committed sequence, eventually.
1043        committed: committed
1044            .assert_has_consistency_of::<Cluster<'a, ClusterTag, EventualConsistency>>(
1045                manual_proof!(
1046                    /** RAFT's election restriction, log-matching, and majority-commit
1047                    rules guarantee every member emits the same committed sequence,
1048                    eventually */
1049                ),
1050            )
1051            .atomic(),
1052        redirected,
1053        leader_views: view_transitions,
1054    }
1055}

1057/// Given a non-consistent stream of events on a Cluster, `raft` will merge them to
1058/// produce an eventually consistent stream containing the union of the events from the
1059/// `requests` stream across all members of the Cluster.
1060///
1061/// The function returns the RAFT log stream, *and* a stream of requests that need to be
1062/// redirected to the leader (paired with the best-known leader hint, `None` while the
1063/// receiving member does not know who leads — e.g. before the leader's first heartbeat
1064/// reaches it; callers wanting a usable hint can hold these themselves).
1065/// The log stream carries full [`LogEntry`]s — the `term` and `index` metadata lets
1066/// consumers deduplicate across leader changes. It is `Atomic` so that user actions
1067/// taken on the commit path complete before the protocol treats entries as committed.
1068///
1069/// # Design: one state machine, one tick
1070///
1071/// Each member runs a single sequential state machine ([`raft_step`]) over the
1072/// complete RAFT state ([`RaftServerState`]) inside one tick, fed by one intra-cluster
1073/// channel. This is a deliberate safety property, not a stylistic choice: RAFT's
1074/// proofs interlock vote decisions and log/ack decisions through shared state
1075/// (`currentTerm`, `votedFor`, the log), and any design that splits that state across
1076/// asynchronously-connected components re-introduces windows where one half acts on a
1077/// stale view of the other. (An earlier two-component design here did exactly that,
1078/// and the simulator reliably reproduced committed-entry truncation; see the
1079/// regression tests `concurrent_elections_never_fork_the_committed_log` and
1080/// `fully_concurrent_run_never_forks_the_committed_log`.)
1081///
1082/// # Timers
1083///
1084/// The two timer streams are inputs so production controls the periods and tests
1085/// control time itself:
1086/// * `election_timer_interrupts`: fire per member; production must randomize the
1087///   period per member (RAFT's tie-breaking).
1088/// * `heartbeat_timer_interrupts`: fire per member with a period much shorter than the
1089///   election timeout, so a live leader's heartbeats arrive between election
1090///   interrupts and suppress them.
1091/// * `net`: builds the fault model for the intra-cluster channel. Simulation tests
1092///   pass `|| TCP.fail_stop().bincode()`; deployments facing partitions pass
1093///   `|| TCP.lossy_delayed_forever().bincode()`, which RAFT is designed to
1094///   tolerate.
1095#[expect(
1096    clippy::type_complexity,
1097    reason = "the return type spells out the exact consistency/ordering guarantees"
1098)]
1099pub fn raft<'a, T, ClusterTag, Con, O, Net>(
1100    requests: Stream<T, Cluster<'a, ClusterTag, Con>, Unbounded, O>,
1101    election_timer_interrupts: Stream<(), Cluster<'a, ClusterTag>>,
1102    heartbeat_timer_interrupts: Stream<(), Cluster<'a, ClusterTag>>,
1103    config: RaftConfig,
1104    net: impl Fn() -> Net,
1105    nondet_order: NonDet,
1106) -> (
1107    Stream<
1108        LogEntry<T>,
1109        Atomic<Cluster<'a, ClusterTag, EventualConsistency>>,
1110        Unbounded,
1111        TotalOrder,
1112    >,
1113    Stream<
1114        (T, Option<MemberId<ClusterTag>>),
1115        Cluster<'a, ClusterTag, NoConsistency>,
1116        Unbounded,
1117        TotalOrder,
1118    >,
1119)
1120where
1121    T: Clone + Serialize + DeserializeOwned + 'a,
1122    O: Ordering,
1123    Con: Consistency,
1124    ClusterTag: 'a,
1125    Net: NetworkFor<RaftRpc<T, ClusterTag>>,
1126    NoOrder: MinOrder<Net::OrderingGuarantee, Min = NoOrder>,
1127{
1128    // The server runs on the consistency-less view of the cluster: `Cluster`'s
1129    // default consistency parameter is `NoConsistency`, so `drop_consistency` is the
1130    // honest (and identity-shaped) conversion from the caller's `Con`-tagged handle.
1131    let cluster = requests.location().drop_consistency();
1132
1133    let outputs = raft_server(
1134        &cluster,
1135        requests.weaken_consistency(),
1136        election_timer_interrupts,
1137        heartbeat_timer_interrupts,
1138        config,
1139        net(),
1140        nondet!(
1141            /// Which member leads, how concurrent requests are interleaved in the
1142            /// log, and when entries replicate fold into the caller's acknowledged
1143            /// non-determinism; the committed sequence itself is consistent.
1144            nondet_order
1145        ),
1146    );
1147
1148    (outputs.committed, outputs.redirected)
1149}

1286    /// every member's view onto the same leader — here driven end-to-end through the
1287    /// real mechanism (the leader's heartbeat timer broadcasting `AppendEntries`):
1288    ///
1289    /// 1. Member 0's election timer fires uncontested and it wins term 1. Before any

1293    ///    one* transition (term untouched — a heartbeat for the current term must not

1295    ///    members hold the identical view `(1, Some(member 0))`.

1297    ///    heartbeat was observed since the last interrupt (no view may change) — and

1472    /// Drives N deterministic [`RaftServerState`] machines connected by an in-memory
1473    /// "network" of per-member inboxes, calling [`raft_step`] directly. Protocol-rule
1474    /// tests run on this instead of the simulator: they need precise, deterministic
1475    /// control over delivery and timers (e.g. "this entry is never replicated"),
1476    /// while schedule exploration is covered by the simulation tests below.
1477    ///
1478    /// Members `deployed..cluster_size` model crashed members: they exist in the
1479    /// protocol configuration (majorities count them) but never take a step, so
1480    /// messages to them pile up unanswered.
1481    struct StepCluster {
1482        states: Vec<RaftServerState<String, Replica>>,
1483        inboxes: Vec<Vec<(MemberId<Replica>, RaftRpc<String, Replica>)>>,
1484        committed: Vec<Vec<LogEntry<String>>>,
1485        redirected: Vec<Vec<(String, Option<MemberId<Replica>>)>>,
1486        cluster_size: usize,
1487        deployed: usize,
1488    }
1489
1490    impl StepCluster {
1491        fn new(deployed: usize, cluster_size: usize) -> Self {
1492            StepCluster {
1493                states: (0..cluster_size).map(|_| RaftServerState::new()).collect(),
1494                inboxes: vec![Vec::new(); cluster_size],
1495                committed: vec![Vec::new(); cluster_size],
1496                redirected: vec![Vec::new(); cluster_size],
1497                cluster_size,
1498                deployed,
1499            }
1500        }
1501
1502        fn id(member: usize) -> MemberId<Replica> {
1503            MemberId::from_raw_id(member as u32)
1504        }
1505
1506        /// Advances `member` by one tick, consuming its inbox plus the given timer
1507        /// events and client requests, and routing its outbound messages.
1508        fn step(&mut self, member: usize, election: bool, heartbeat: bool, requests: &[&str]) {
1509            let messages = std::mem::take(&mut self.inboxes[member]);
1510            let output = raft_step(
1511                &mut self.states[member],
1512                RaftStepInput {
1513                    me: Self::id(member),
1514                    other_members: (0..self.cluster_size)
1515                        .filter(|target| *target != member)
1516                        .map(Self::id)
1517                        .collect(),
1518                    cluster_size: self.cluster_size,
1519                    election_timer_fired: election,
1520                    heartbeat_timer_fired: heartbeat,
1521                    requests: requests
1522                        .iter()
1523                        .map(|request| (*request).to_owned())
1524                        .collect(),
1525                    messages,
1526                },
1527            );
1528            for (target, message) in output.outbound {
1529                self.inboxes[target.get_raw_id() as usize].push((Self::id(member), message));
1530            }
1531            self.committed[member].extend(output.committed);
1532            self.redirected[member].extend(output.redirected);
1533        }
1534
1535        /// Steps deployed members (no timers) until no inbox holds a message —
1536        /// i.e. all in-flight traffic, including replies to replies, has settled.
1537        fn deliver_until_quiet(&mut self) {
1538            loop {
1539                let mut progressed = false;
1540                for member in 0..self.deployed {
1541                    if !self.inboxes[member].is_empty() {
1542                        progressed = true;
1543                        self.step(member, false, false, &[]);
1544                    }
1545                }
1546                if !progressed {
1547                    return;
1548                }
1549            }
1550        }
1551
1552        /// Runs a real election at `member`: fires its election timer (twice if the
1553        /// first interrupt is consumed by heartbeat suppression), settles the vote
1554        /// traffic, and asserts the member won. Returns the won term.
1555        fn elect(&mut self, member: usize) -> usize {
1556            // At most 2 interrupts: the first may be consumed by heartbeat
1557            // suppression; the second must start a candidacy.
1558            let mut attempts = 0;
1559            while self.states[member].role == RaftState::Follower {
1560                assert!(
1561                    attempts < 2,
1562                    "member {}'s election timer never started a candidacy \
1563                     after {} interrupts",
1564                    member,
1565                    attempts
1566                );
1567                self.step(member, true, false, &[]);
1568                attempts += 1;
1569            }
1570            self.deliver_until_quiet();
1571            assert!(
1572                self.states[member].role == RaftState::Leader,
1573                "member {} failed to win its election (role {:?}, term {})",
1574                member,
1575                self.states[member].role,
1576                self.states[member].term
1577            );
1578            self.states[member].term
1579        }
1580
1581        /// Installs `member` as the leader of `term` directly, modeling figure 8's
1582        /// crash-and-reelect histories without simulating the message loss that
1583        /// produces them (exactly what the pre-unification sim tests did by
1584        /// injecting `LeaderView`s). The state produced is a protocol-reachable
1585        /// leadership: term advanced, vote self-recorded, follower bookkeeping reset.
1586        fn crown(&mut self, member: usize, term: usize) {
1587            assert!(term > self.states[member].term, "terms only move forward");
1588            let log_len = self.states[member].log.len();
1589            let state = &mut self.states[member];
1590            state.term = term;
1591            state.role = RaftState::Leader;
1592            state.voted_for = Some(Self::id(member));
1593            state.votes.clear();
1594            state.known_leader = None;
1595            state.next_index.clear();
1596            state.match_index.clear();
1597            for target in 0..self.cluster_size {
1598                if target != member {
1599                    state.next_index.insert(Self::id(target), log_len + 1);
1600                    state.match_index.insert(Self::id(target), 0);
1601                }
1602            }
1603        }
1604
1605        /// Sends `payload` to `member` and asserts it was appended as leader rather
1606        /// than redirected.
1607        fn submit(&mut self, member: usize, payload: &str) {
1608            let redirects_before = self.redirected[member].len();
1609            self.step(member, false, false, &[payload]);
1610            assert_eq!(
1611                self.redirected[member].len(),
1612                redirects_before,
1613                "member {} redirected \"{}\" instead of appending it as leader",
1614                member,
1615                payload
1616            );
1617        }
1618
1619        /// One settled heartbeat round: the leader broadcasts, and all resulting
1620        /// traffic (appends, acks, and the leader's reactions) quiesces.
1621        fn heartbeat_round(&mut self, leader: usize) {
1622            self.step(leader, false, true, &[]);
1623            self.deliver_until_quiet();
1624        }

1626        fn assert_committed(&self, expected: &[(&str, usize)]) {

1629                .enumerate()

1657    /// The interlock that motivated unifying the protocol state into one step (the
1658    /// cross-tick race fixed by this design; see [`raft`]'s docs): a member's vote
1659    /// decisions and its log/ack decisions must be read-modify-writes against the
1660    /// same state, in both directions, even when the triggering messages arrive in
1661    /// the same batch.
1662    ///
1663    /// * **Ack-then-vote**: a member that has acknowledged an entry compares later
1664    ///   vote requests against the log *including* that entry, so a candidate
1665    ///   missing it is refused (§5.4.1 reads the live log).
1666    /// * **Vote-then-ack**: a member that has voted in a higher term rejects an
1667    ///   old-term leader's `AppendEntries` outright, so its ack can never help
1668    ///   commit an entry behind the new leader's back (§5.1 reads the live term).
1669    #[test]
1670    fn vote_and_ack_decisions_interlock() {
1671        let leader_0 = StepCluster::id(0);
1672        let candidate_2 = StepCluster::id(2);
1673        let entry = |index: usize| LogEntry {
1674            message: format!("entry-{}", index),
1675            term_received: 1,
1676            index,
1677        };
1678        let append = |entries: Vec<LogEntry<String>>, prev: usize| {
1679            RaftRpc::AppendEntries(AppendEntriesRequest {
1680                term: 1,
1681                leader: leader_0.clone(),
1682                prev_log_index: prev,
1683                prev_log_term: if prev == 0 { 0 } else { 1 },
1684                entries,
1685                leader_commit: 0,
1686            })
1687        };
1688
1689        // Ack-then-vote: member 1 acknowledges entry 1 from the term-1 leader, and
1690        // *in the same batch* receives a term-2 vote request from a candidate whose
1691        // log is empty. Whatever the processing order, the two decisions must never
1692        // combine into "acked the entry AND endorsed a candidate missing it": the
1693        // canonical order processes the vote first (empty-vs-empty log: granted,
1694        // term becomes 2), which forces the term-1 append to be rejected.
1695        let mut state: RaftServerState<String, Replica> = RaftServerState::new();
1696        let output = raft_step(
1697            &mut state,
1698            RaftStepInput {
1699                me: StepCluster::id(1),
1700                other_members: vec![StepCluster::id(0), StepCluster::id(2)],
1701                cluster_size: 3,
1702                election_timer_fired: false,
1703                heartbeat_timer_fired: false,
1704                requests: vec![],
1705                messages: vec![
1706                    (leader_0.clone(), append(vec![entry(1)], 0)),
1707                    (
1708                        candidate_2.clone(),
1709                        RaftRpc::RequestVote(RequestVoteDto {
1710                            term: 2,
1711                            last_log_index: 0,
1712                            last_log_term: 0,
1713                        }),
1714                    ),
1715                ],
1716            },
1717        );
1718        assert_eq!(state.term, 2, "the vote request's term must be adopted");
1719        assert_eq!(
1720            state.voted_for,
1721            Some(candidate_2.clone()),
1722            "the empty-log candidate is legitimately granted against an empty log"
1723        );
1724        assert!(
1725            state.log.is_empty(),
1726            "having endorsed a term-2 candidate, the term-1 append must be rejected"
1727        );
1728        let acked = output.outbound.iter().any(|(_, message)| {
1729            matches!(
1730                message,
1731                RaftRpc::AppendEntriesReply(reply) if reply.success
1732            )
1733        });
1734        assert!(
1735            !acked,
1736            "an ack and a vote for a candidate missing the acked entry must never \
1737             both happen"
1738        );
1739
1740        // Vote-then-ack, later ticks: the vote persists, so an old-term append in a
1741        // *subsequent* tick is still rejected (the term was adopted atomically with
1742        // the grant — there is no window where the log half still believes term 1).
1743        let output = raft_step(
1744            &mut state,
1745            RaftStepInput {
1746                me: StepCluster::id(1),
1747                other_members: vec![StepCluster::id(0), StepCluster::id(2)],
1748                cluster_size: 3,
1749                election_timer_fired: false,
1750                heartbeat_timer_fired: false,
1751                requests: vec![],
1752                messages: vec![(leader_0.clone(), append(vec![entry(1)], 0))],
1753            },
1754        );
1755        assert!(
1756            state.log.is_empty(),
1757            "a stale-term append must stay rejected"
1758        );
1759        assert!(matches!(
1760            &output.outbound[..],
1761            [(target, RaftRpc::AppendEntriesReply(reply))]
1762                if *target == leader_0 && !reply.success && reply.term == 2
1763        ));
1764
1765        // Ack-then-vote, later ticks: a member whose log holds an acknowledged entry
1766        // refuses a later candidate that lacks it (§5.4.1 against the live log).
1767        let mut state: RaftServerState<String, Replica> = RaftServerState::new();
1768        raft_step(
1769            &mut state,
1770            RaftStepInput {
1771                me: StepCluster::id(1),
1772                other_members: vec![StepCluster::id(0), StepCluster::id(2)],
1773                cluster_size: 3,
1774                election_timer_fired: false,
1775                heartbeat_timer_fired: false,
1776                requests: vec![],
1777                messages: vec![(leader_0.clone(), append(vec![entry(1)], 0))],
1778            },
1779        );
1780        assert_eq!(
1781            state.log.len(),
1782            1,
1783            "the term-1 append is accepted this time"
1784        );
1785        raft_step(
1786            &mut state,
1787            RaftStepInput {
1788                me: StepCluster::id(1),
1789                other_members: vec![StepCluster::id(0), StepCluster::id(2)],

2350        let mut flow = FlowBuilder::new();
2351        let cluster = flow.cluster::<Replica>();
2352
2353        let (election_interrupt_send, election_timer_interrupts) = cluster.sim_input();
2354        let (heartbeat_interrupt_send, heartbeat_timer_interrupts) = cluster.sim_input();
2355        let (request_send, requests) = cluster.sim_input::<String, _, _>();
2356
2357        let (committed, redirected) = raft(
2358            requests,
2359            election_timer_interrupts,
2360            heartbeat_timer_interrupts,
2361            RaftConfig { cluster_size: N },
2362            || TCP.fail_stop().bincode(),
2363            nondet!(
2364                /** which member leads and how concurrent requests are ordered is
2365                non-deterministic; the committed sequence must not be */
2366            ),
2367        );
2368
2369        let committed_recv = committed.end_atomic().sim_cluster_output();
2370        let redirected_recv = redirected.sim_cluster_output();
2371
2372        // The committed stream carries a consistency assertion (see log_replication);
2373        // the simulator cannot validate those yet and asks tests to skip them.
2374        flow.sim()
2375            .skip_consistency_assertions()
2376            .with_cluster_size(&cluster, N)
2377            .fuzz(async || {
2378                let mut committed: Vec<Vec<LogEntry<String>>> = vec![Vec::new(); N];
2379
2380                // Quiesce (phase barrier), fold each member's newly committed entries
2381                // into its history, drain redirects, and check the fork invariant.
2382                let collect_and_check = async |committed: &mut Vec<Vec<LogEntry<String>>>| {
2383                    hydro_lang::sim::quiesce().await;
2384                    for member in 0..N as u32 {
2385                        committed[member as usize]
2386                            .extend(committed_recv.collect::<Vec<_>>(member).await);
2387                        let _: Vec<(String, Option<MemberId<Replica>>)> =
2388                            redirected_recv.collect(member).await;
2389                    }
2390                    assert_prefix_consistent(committed);
2391                };
2392
2393                // Phase 1: member 0 wins term 1 uncontested and commits a seed entry,
2394                // so later (possibly stale) challengers have something to be behind.
2395                // The `quiesce` barrier (in collect_and_check) after the interrupt lets
2396                // the election settle
2397                // (vote traffic and view propagation) before the request is sent —
2398                // otherwise the request could race ahead of member 0's leadership and
2399                // be redirected into the void. This phase is deliberately race-free,
2400                // so a liveness bound is safe (same rationale as
2401                // composed_raft_elects_replicates_and_suppresses).
2402                election_interrupt_send.send(0, ());
2403                collect_and_check(&mut committed).await;
2404                request_send.send(0, "seed".to_owned());
2405                let mut pumps = 0;
2406                while committed.iter().any(|entries| entries.is_empty()) {
2407                    assert!(
2408                        pumps < MAX_ROUNDS,
2409                        "seed entry did not commit within {} uncontested heartbeat \
2410                         rounds; per-member counts: {:?}",
2411                        MAX_ROUNDS,
2412                        committed
2413                            .iter()
2414                            .map(|entries| entries.len())
2415                            .collect::<Vec<_>>()
2416                    );
2417                    pumps += 1;
2418                    heartbeat_interrupt_send.send(0, ());
2419                    collect_and_check(&mut committed).await;
2420                }
2421
2422                // Phase 2: racy rounds. Everything inside a burst is sent without
2423                // quiescing in between, so replication of a fresh entry and a
2424                // challenger's candidacy overlap freely.
2425                for round in 0..RACY_ROUNDS {
2426                    let challenger = 1 + (round % (N - 1)) as u32;
2427
2428                    // Prime: consume the challenger's heartbeat-suppression flag so
2429                    // the burst's second interrupt can actually start a candidacy
2430                    // (unless a racing heartbeat re-arms it — also a valid schedule).
2431                    election_interrupt_send.send(challenger, ());
2432
2433                    // Burst: fresh entry + replication pumps + concurrent candidacy.
2434                    request_send.send(0, format!("racy-{}", round));
2435                    request_send.send(challenger, format!("challenger-{}", round));
2436                    heartbeat_interrupt_send.send(0, ());
2437                    election_interrupt_send.send(challenger, ());
2438                    heartbeat_interrupt_send.send(0, ());
2439                    collect_and_check(&mut committed).await;
2440
2441                    // Settle: pump every member's heartbeat timer (only an acting
2442                    // leader — whoever that now is — emits AppendEntries), letting
2443                    // any successful election finish replicating so the next round
2444                    // starts from a coherent, but unspecified, leadership.
2445                    for _ in 0..3 {
2446                        for member in 0..N as u32 {
2447                            heartbeat_interrupt_send.send(member, ());
2448                        }
2449                        collect_and_check(&mut committed).await;
2450                    }
2451                }
2452            });
2453    }

2455    /// **Regression net for the fixed cross-tick race** (same underlying bug as
2456    /// [`concurrent_elections_never_fork_the_committed_log`], where the history is
2457    /// documented; before the fix this reproduced it in 2 of 3 runs).
2458    ///
2459    /// The maximal version of the no-quiescence idea: *every* input for the entire
2460    /// run — election timer interrupts, client requests, and heartbeat pumps, for
2461    /// every member — is sent up front with **no intermediate quiescence at all**.
2462    /// The simulator's quiescing `collect` is used exactly once, at the end, to
2463    /// drain outputs; by then it cannot constrain how any inputs interleaved. The
2464    /// fuzzer therefore owns the complete schedule: which timer fires between which
2465    /// message deliveries, how batches form on each member's tick, everything.
2466    /// For deeper exploration, run it under coverage-guided fuzzing:
2467    /// `cargo sim -- fully_concurrent_run` (a found failure is saved as a
2468    /// minimized reproducer under `src/cluster/sim-failures/`, which plain
2469    /// `cargo test` then replays deterministically).
2470    ///
2471    /// The price of zero barriers is that nothing intermediate can be asserted — no
2472    /// "member X won term Y", and no liveness at all (a schedule where elections
2473    /// perpetually contend and nothing ever commits is legal). What remains is the
2474    /// protocol's core safety contract, checked at the end:
2475    ///
2476    /// 1. each member's committed stream has contiguous log indexes from 1 (each
2477    ///    position emitted exactly once, in order);
2478    /// 2. all members' committed histories are pairwise prefix-consistent (lagging
2479    ///    is fine, forking is not);
2480    /// 3. [`raft_step`]'s truncation guard never fires (surfaces as a panic /
2481    ///    SIGABRT if a committed entry is overwritten).
2482    ///
2483    /// Compared to the targeted test above, this explores a much larger schedule
2484    /// space with no phase structure biasing it toward the historical window — the
2485    /// stronger long-term net, while the targeted test remains the faster, more
2486    /// diagnostic one.
2487    #[test]
2488    fn fully_concurrent_run_never_forks_the_committed_log() {
2489        const N: usize = 3;
2490        const ELECTIONS_PER_MEMBER: usize = 2;
2491        const HEARTBEAT_PUMPS_PER_MEMBER: usize = 6;
2492
2493        let mut flow = FlowBuilder::new();
2494        let cluster = flow.cluster::<Replica>();
2495
2496        let (election_interrupt_send, election_timer_interrupts) = cluster.sim_input();
2497        let (heartbeat_interrupt_send, heartbeat_timer_interrupts) = cluster.sim_input();
2498        let (request_send, requests) = cluster.sim_input::<String, _, _>();
2499
2500        let (committed, redirected) = raft(
2501            requests,
2502            election_timer_interrupts,
2503            heartbeat_timer_interrupts,
2504            RaftConfig { cluster_size: N },
2505            || TCP.fail_stop().bincode(),
2506            nondet!(
2507                /** which member leads and how concurrent requests are ordered is
2508                non-deterministic; the committed sequence must not be */
2509            ),
2510        );
2511
2512        let committed_recv = committed.end_atomic().sim_cluster_output();
2513        let redirected_recv = redirected.sim_cluster_output();
2514
2515        // The committed stream carries a consistency assertion (see log_replication);
2516        // the simulator cannot validate those yet and asks tests to skip them.
2517        flow.sim()
2518            .skip_consistency_assertions()
2519            .with_cluster_size(&cluster, N)
2520            .fuzz(async || {
2521                // Fire everything up front. No `.await` anywhere in this block, so
2522                // all inputs are concurrently outstanding and the fuzzer alone
2523                // decides every interleaving. Sends are woven round-robin across
2524                // members and input kinds only to avoid biasing the schedule via
2525                // enqueue order.
2526                for wave in 0..ELECTIONS_PER_MEMBER {
2527                    for member in 0..N as u32 {
2528                        election_interrupt_send.send(member, ());
2529                        request_send.send(member, format!("wave-{}-member-{}", wave, member));
2530                    }
2531                }
2532                for _ in 0..HEARTBEAT_PUMPS_PER_MEMBER {
2533                    for member in 0..N as u32 {
2534                        heartbeat_interrupt_send.send(member, ());
2535                    }
2536                }
2537
2538                // The single, final quiescence: drain both outputs.
2539                let mut committed: Vec<Vec<LogEntry<String>>> = vec![Vec::new(); N];
2540                for member in 0..N as u32 {
2541                    committed[member as usize]
2542                        .extend(committed_recv.collect::<Vec<_>>(member).await);
2543                    let _: Vec<(String, Option<MemberId<Replica>>)> =
2544                        redirected_recv.collect(member).await;
2545                }
2546
2547                // Safety only. (1) Per-member: contiguous indexes from 1, in order.
2548                for (member, history) in committed.iter().enumerate() {
2549                    for (position, entry) in history.iter().enumerate() {
2550                        assert_eq!(
2551                            entry.index,
2552                            position + 1,
2553                            "member {} emitted committed entries out of order or \
2554                             with gaps: {:?}",
2555                            member,
2556                            history.iter().map(|e| e.index).collect::<Vec<_>>()
2557                        );
2558                    }
2559                }
2560                // (2) Pairwise: histories may lag but never fork.
2561                for a in 0..N {
2562                    for b in (a + 1)..N {
2563                        for (position, (entry_a, entry_b)) in
2564                            committed[a].iter().zip(&committed[b]).enumerate()
2565                        {
2566                            assert_eq!(
2567                                entry_a, entry_b,
2568                                "committed logs forked: members {} and {} disagree \
2569                                 at committed position {}",
2570                                a, b, position
2571                            );
2572                        }
2573                    }
2574                }
2575            });
2576    }
