1use hydro_lang::live_collections::stream::{AtLeastOnce, NoOrder};
2use hydro_lang::prelude::*;
3
4pub fn capitalize<'a>(input: Stream<String, Process<'a, ()>>) {
5    input
6        .map(q!(|s| s.to_uppercase()))
7        .embedded_output("output");
8}
9
10/// Like [`capitalize`], but emits an unordered, at-least-once stream, which `embedded_output`
11/// also accepts.
12pub fn capitalize_unordered<'a>(input: Stream<String, Process<'a, ()>>) {
13    input
14        .map(q!(|s| s.to_uppercase()))
15        .weaken_ordering::<NoOrder>()
16        .weaken_retries::<AtLeastOnce>()
17        .embedded_output("output");
18}
