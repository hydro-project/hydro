1use hydro_lang::location::MemberId;
2use hydro_lang::prelude::*;
3
4pub struct Src {}
5pub struct Dst {}
6
7/// A payload that deliberately does **not** implement `Serialize`/`DeserializeOwned`.
8///
9/// `.embedded()` channels move values in-process without ever serializing them, so demuxing
10/// this type must compile without serde impls (regression test for hydro-project/hydro#3158).
11pub struct OpaquePayload {
12    pub text: String,
13}
14
15/// Like [`super::o2m_broadcast::o2m_broadcast`], but demuxes to explicit members with
16/// `.embedded()` serialization and a payload type that has no serde derives.
17pub fn o2m_demux_embedded<'a>(
18    cluster: &Cluster<'a, Dst>,
19    input: Stream<String, Process<'a, Src>>,
20) -> Stream<String, Cluster<'a, Dst>> {
21    input
22        .map(q!(|s| (
23            MemberId::from_raw_id(0),
24            OpaquePayload { text: s }
25        )))
26        .demux(cluster, TCP.fail_stop().embedded().name("demux_data"))
27        .map(q!(|payload| payload.text.to_uppercase()))
28}
