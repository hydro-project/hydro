1use hydro_lang::live_collections::stream::TotalOrder;
2use hydro_lang::location::MemberId;
3use hydro_lang::prelude::*;
4
5pub struct Client;
6pub struct LogServer;
7
8/// Builds a totally-ordered event log on a central server from events emitted
9/// by a cluster of clients. Events from *different* clients may interleave
10/// arbitrarily (the arrival order is non-deterministic), but the log always
11/// preserves each client's local order, since TCP delivers an in-order prefix
12/// per sender.
13///
14/// This example is featured on the Hydro website landing page.
15pub fn ordered_event_log<'a>(
16    events: Stream<String, Cluster<'a, Client>>,
17    server: &Process<'a, LogServer>,
18) -> Stream<(MemberId<Client>, String), Process<'a, LogServer>, Unbounded, TotalOrder> {
19    events
20        .send(server, TCP.fail_stop().bincode())
21        .entries_partially_ordered(nondet!(/** log order = arrival order */))
22}
23
24#[cfg(test)]
25mod tests {
26    use super::*;
27
28    #[test]
29    fn test_per_member_prefix_order() {
30        let mut flow = FlowBuilder::new();
31        let clients = flow.cluster::<Client>();
32        let server = flow.process::<LogServer>();
33
34        let (event_port, events) = clients.sim_input();
35        let log = ordered_event_log(events, &server).sim_output();
36
37        let instances = flow
38            .sim()
39            .with_cluster_size(&clients, 3)
40            .exhaustive(async || {
41                event_port.send(0, "a1".to_owned());
42                event_port.send(0, "a2".to_owned());
43                event_port.send(1, "b1".to_owned());
44
45                let entries: Vec<_> = log.collect().await;
46                let pos = |m: &str| entries.iter().position(|(_, e)| e == m).unwrap();
47                // events interleave arbitrarily, but per-client order always holds
48                assert!(pos("a1") < pos("a2"));
49            });
50
51        // The landing page displays this exact exploration: three distinct
52        // interleavings of [a1, a2] x [b1] that preserve per-member prefixes.
53        assert_eq!(instances, 3);
54    }
55}
